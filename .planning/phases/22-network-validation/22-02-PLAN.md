Implement MessageOrderer for message ordering per channel:

```rust
use std::collections::{HashMap, BTreeMap};
use parking_lot::Mutex;
use crate::proto::chat_v1::Message;

/// Maximum number of out-of-order messages to buffer per channel.
const MAX_BUFFER_SIZE: usize = 100;

/// Maximum time to wait for missing messages (in milliseconds).
const MAX_BUFFER_AGE_MS: u64 = 30_000;  // 30 seconds

/// Ordering result for a message.
#[derive(Debug)]
pub enum OrderingResult {
    /// Message is in order, ready for processing.
    Ready(Message),
    /// Message is out of order, buffered for later.
    Buffered,
    /// Message is a duplicate (same timestamp seen).
    Duplicate,
    /// Buffer is full, message dropped.
    Dropped,
}

/// Per-channel ordering state.
struct ChannelOrderState {
    /// Last processed timestamp for this channel.
    last_timestamp: u64,
    /// Buffered out-of-order messages, keyed by timestamp.
    /// Using BTreeMap for automatic ordering by timestamp.
    buffer: BTreeMap<u64, Message>,
    /// When the oldest buffered message was received.
    oldest_buffer_time: Option<std::time::Instant>,
}

pub struct MessageOrderer {
    /// Per-channel ordering state.
    channels: Mutex<HashMap<String, ChannelOrderState>>,
    /// Maximum buffer size per channel.
    max_buffer_size: usize,
}

impl MessageOrderer {
    pub fn new() -> Self;
    pub fn with_buffer_size(max_buffer_size: usize) -> Self;

    /// Process a message for ordering.
    /// Returns Ready if in order, Buffered if out of order.
    pub fn order(&self, message: Message) -> OrderingResult;

    /// Get any messages that are now ready (after a message was processed).
    /// Call this after processing a Ready message to get buffered messages.
    pub fn get_ready(&self, channel_id: &str) -> Vec<Message>;

    /// Flush stale buffered messages (older than MAX_BUFFER_AGE_MS).
    /// Returns messages that were flushed (for logging/processing anyway).
    pub fn flush_stale(&self) -> Vec<Message>;

    /// Get ordering statistics.
    pub fn stats(&self) -> OrdererStats;
}

pub struct OrdererStats {
    pub channels_tracked: usize,
    pub total_buffered: usize,
}

impl Default for MessageOrderer {
    fn default() -> Self {
        Self::new()
    }
}
```

Key behaviors:
- Track last_timestamp per channel
- If message.timestamp > last_timestamp: in order, update last_timestamp
- If message.timestamp <= last_timestamp: out of order, buffer it
- When processing Ready message, check if any buffered messages are now ready
- Flush stale messages periodically (caller's responsibility to call flush_stale)

Use timestamp for ordering (proto Message has `timestamp: u64` in milliseconds). Messages with same timestamp are considered duplicates at orderer level.

Do NOT make this async - ordering is fast in-memory operation.

```rust
use crate::proto::chat_v1::Message;
use super::deduplicator::MessageDeduplicator;
use super::orderer::{MessageOrderer, OrderingResult};

/// Result of pipeline processing.
#[derive(Debug)]
pub enum PipelineResult {
    /// Message passed validation and is ready for processing.
    Ready(Message),
    /// Message is valid but buffered (out of order).
    Buffered,
    /// Message was a duplicate, already processed.
    Duplicate,
    /// Message failed validation.
    Invalid(ValidationError),
    /// Message was dropped (buffer full).
    Dropped,
}

/// Unified validation pipeline for incoming messages.
///
/// Processing order:
/// 1. Deduplication check (by message ID)
/// 2. Content validation (format, size, timestamp)
/// 3. Ordering (per-channel sequence)
pub struct MessageValidationPipeline {
    deduplicator: MessageDeduplicator,
    validator: ContentValidator,
    orderer: MessageOrderer,
}

impl MessageValidationPipeline {
    pub fn new() -> Self {
        Self {
            deduplicator: MessageDeduplicator::with_default_capacity(),
            validator: ContentValidator::with_defaults(),
            orderer: MessageOrderer::new(),
        }
    }

    pub fn with_config(
        dedup_capacity: usize,
        validation_limits: ValidationLimits,
        orderer_buffer_size: usize,
    ) -> Self;

    /// Process a message through the full validation pipeline.
    pub fn process(&self, message: Message) -> PipelineResult {
        // 1. Check for duplicate by message ID
        if self.deduplicator.is_duplicate(&message.id) {
            return PipelineResult::Duplicate;
        }

        // 2. Validate content
        if let Err(e) = self.validator.validate(&message) {
            return PipelineResult::Invalid(e);
        }

        // 3. Order the message
        match self.orderer.order(message) {
            OrderingResult::Ready(msg) => {
                // Mark as processed only when ready
                self.deduplicator.mark_processed(&msg.id);
                PipelineResult::Ready(msg)
            }
            OrderingResult::Buffered => PipelineResult::Buffered,
            OrderingResult::Duplicate => PipelineResult::Duplicate,
            OrderingResult::Dropped => PipelineResult::Dropped,
        }
    }

    /// Get messages that are now ready after processing.
    pub fn get_ready(&self, channel_id: &str) -> Vec<Message> {
        let ready = self.orderer.get_ready(channel_id);
        // Mark newly ready messages as processed
        for msg in &ready {
            self.deduplicator.mark_processed(&msg.id);
        }
        ready
    }

    /// Flush stale buffered messages.
    pub fn flush_stale(&self) -> Vec<Message> {
        let stale = self.orderer.flush_stale();
        for msg in &stale {
            self.deduplicator.mark_processed(&msg.id);
        }
        stale
    }

    /// Get combined statistics.
    pub fn stats(&self) -> PipelineStats;
}

pub struct PipelineStats {
    pub deduplicator: super::deduplicator::DeduplicatorStats,
    pub orderer: super::orderer::OrdererStats,
}

impl Default for MessageValidationPipeline {
    fn default() -> Self {
        Self::new()
    }
}
```
