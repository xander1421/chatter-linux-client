use std::collections::{HashSet, VecDeque};
use std::hash::Hash;

pub struct BoundedHashSet<T> {
    set: HashSet<T>,
    order: VecDeque<T>,  // Tracks insertion order for LRU eviction
    capacity: usize,
}

impl<T: Hash + Eq + Clone> BoundedHashSet<T> {
    pub fn new(capacity: usize) -> Self;
    pub fn contains(&self, value: &T) -> bool;
    pub fn insert(&mut self, value: T) -> bool;  // Returns true if new
    pub fn len(&self) -> usize;
    pub fn is_empty(&self) -> bool;
    pub fn capacity(&self) -> usize;
}
```

Key behaviors:
- When at capacity, remove oldest entry before inserting new
- insert() returns false if already present (duplicate detection)
- Use VecDeque for O(1) removal from front
- Clone constraint needed for order tracking

Do NOT use fancy eviction strategies - simple FIFO-based eviction is sufficient for deduplication where we just need to remember "recently seen" GUIDs.

use thiserror::Error;
use crate::proto::chat_v1::Message;

#[derive(Debug, Clone, Error)]
pub enum ValidationError {
    #[error("missing required field: {0}")]
    MissingField(String),

    #[error("invalid message ID format")]
    InvalidMessageId,

    #[error("invalid channel ID format")]
    InvalidChannelId,

    #[error("timestamp out of range: {0}")]
    InvalidTimestamp(u64),

    #[error("content too large: {size} bytes (max {max})")]
    ContentTooLarge { size: usize, max: usize },

    #[error("empty message content")]
    EmptyContent,
}

/// Content limits for validation.
pub struct ValidationLimits {
    /// Maximum message content size in bytes (default: 64KB)
    pub max_content_size: usize,
    /// Maximum timestamp drift from now in seconds (default: 5 minutes)
    pub max_timestamp_drift_secs: u64,
    /// Minimum valid timestamp (Unix epoch 2020-01-01)
    pub min_timestamp: u64,
}

impl Default for ValidationLimits {
    fn default() -> Self {
        Self {
            max_content_size: 64 * 1024,  // 64KB
            max_timestamp_drift_secs: 5 * 60,  // 5 minutes
            min_timestamp: 1577836800000,  // 2020-01-01 in millis
        }
    }
}

pub struct ContentValidator {
    limits: ValidationLimits,
}

impl ContentValidator {
    pub fn new(limits: ValidationLimits) -> Self;
    pub fn with_defaults() -> Self;

    /// Validate a message, returning error if invalid.
    pub fn validate(&self, message: &Message) -> Result<(), ValidationError>;

    /// Validate message ID format (non-empty, reasonable length).
    fn validate_id(&self, id: &str, field_name: &str) -> Result<(), ValidationError>;

    /// Validate timestamp is within reasonable bounds.
    fn validate_timestamp(&self, timestamp: u64) -> Result<(), ValidationError>;
}
```

Validation rules:
- message.id must be non-empty and <= 64 chars
- message.channel_id must be non-empty and <= 64 chars
- timestamp must be > min_timestamp and not too far in future
- Content size check (if content is present)

Do NOT validate author_id as empty - sealed sender messages have no author (per Signal protocol).

Update mod.rs to export ValidationError and ValidationLimits.
