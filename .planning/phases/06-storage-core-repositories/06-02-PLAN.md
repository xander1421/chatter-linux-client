```sql
-- Channels
CREATE TABLE channels (
    id TEXT PRIMARY KEY,
    channel_type INTEGER NOT NULL,
    name TEXT NOT NULL,
    description TEXT,
    image_url TEXT,
    owner_id TEXT NOT NULL,
    -- Config fields (flattened for simplicity)
    persistence INTEGER NOT NULL DEFAULT 0,
    max_members INTEGER,
    default_ttl_seconds INTEGER,
    allow_reactions INTEGER NOT NULL DEFAULT 1,
    allow_threads INTEGER NOT NULL DEFAULT 1,
    allow_attachments INTEGER NOT NULL DEFAULT 1,
    require_encryption INTEGER NOT NULL DEFAULT 1,
    -- Timestamps and state
    created_at TEXT NOT NULL,
    updated_at TEXT NOT NULL,
    archived INTEGER NOT NULL DEFAULT 0,
    unread_count INTEGER NOT NULL DEFAULT 0,
    last_message_at TEXT
);
CREATE INDEX idx_channels_owner ON channels(owner_id);
CREATE INDEX idx_channels_last_message ON channels(last_message_at DESC);

-- Channel memberships
CREATE TABLE memberships (
    channel_id TEXT NOT NULL REFERENCES channels(id) ON DELETE CASCADE,
    user_id TEXT NOT NULL,
    role INTEGER NOT NULL DEFAULT 0,
    joined_at TEXT NOT NULL,
    last_read_message_id TEXT,
    is_muted INTEGER NOT NULL DEFAULT 0,
    muted_until TEXT,
    notification_preference INTEGER NOT NULL DEFAULT 0,
    PRIMARY KEY (channel_id, user_id)
);
CREATE INDEX idx_memberships_user ON memberships(user_id);
```

Enum mappings:
- ChannelType: 0=Public, 1=Private, 2=DirectMessage
- MessagePersistence: 0=Permanent, 1=Ephemeral, 2=Hybrid
- MemberRole: 0=Member, 1=Moderator, 2=Admin, 3=Owner
- NotificationPreference: 0=All, 1=Mentions, 2=None
