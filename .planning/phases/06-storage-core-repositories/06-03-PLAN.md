```sql
-- Messages
CREATE TABLE messages (
    id TEXT PRIMARY KEY,
    channel_id TEXT NOT NULL REFERENCES channels(id) ON DELETE CASCADE,
    sender_id TEXT,  -- NULL for sealed sender
    -- Content stored as JSON (MessageContent is complex enum)
    content_type INTEGER NOT NULL,  -- Discriminant for quick filtering
    content_json TEXT NOT NULL,     -- Full serialized MessageContent
    thread_id TEXT,
    reply_to TEXT,
    -- Attachments stored as JSON array of UUIDs
    attachments_json TEXT NOT NULL DEFAULT '[]',
    is_pinned INTEGER NOT NULL DEFAULT 0,
    is_edited INTEGER NOT NULL DEFAULT 0,
    is_deleted INTEGER NOT NULL DEFAULT 0,
    created_at TEXT NOT NULL,
    updated_at TEXT NOT NULL,
    edited_at TEXT,
    -- Reaction data as JSON (for local caching)
    reaction_counts_json TEXT NOT NULL DEFAULT '[]',
    current_user_reactions_json TEXT NOT NULL DEFAULT '[]',
    -- AI-related fields
    mentioned_integrations_json TEXT NOT NULL DEFAULT '[]',
    targets_ai_bot INTEGER NOT NULL DEFAULT 0
);

-- Indexes for common queries
CREATE INDEX idx_messages_channel_created ON messages(channel_id, created_at DESC);
CREATE INDEX idx_messages_thread ON messages(thread_id) WHERE thread_id IS NOT NULL;
CREATE INDEX idx_messages_sender ON messages(sender_id) WHERE sender_id IS NOT NULL;
CREATE INDEX idx_messages_pinned ON messages(channel_id) WHERE is_pinned = 1;

-- Threads metadata (derived/cached)
CREATE TABLE threads (
    root_message_id TEXT PRIMARY KEY REFERENCES messages(id) ON DELETE CASCADE,
    channel_id TEXT NOT NULL REFERENCES channels(id) ON DELETE CASCADE,
    reply_count INTEGER NOT NULL DEFAULT 0,
    last_reply_at TEXT,
    participant_ids_json TEXT NOT NULL DEFAULT '[]'
);
CREATE INDEX idx_threads_channel ON threads(channel_id);
```

content_type discriminant mapping (for filtering without JSON parsing):

```rust
impl MessageRepository {
    pub fn new(db: Arc<Database>) -> Self

    // Message CRUD
    pub fn insert(&self, message: &Message) -> Result<(), StorageError>
    pub fn get(&self, id: &MessageId) -> Result<Option<Message>, StorageError>
    pub fn update(&self, message: &Message) -> Result<(), StorageError>
    pub fn delete(&self, id: &MessageId) -> Result<(), StorageError>

    // Queries
    pub fn list_for_channel(
        &self,
        channel_id: &ChannelId,
        before: Option<&MessageId>,  // For pagination (cursor-based)
        limit: u32,
    ) -> Result<Vec<Message>, StorageError>

    pub fn list_for_thread(
        &self,
        thread_id: &MessageId,
        before: Option<&MessageId>,
        limit: u32,
    ) -> Result<Vec<Message>, StorageError>

    pub fn list_pinned(&self, channel_id: &ChannelId) -> Result<Vec<Message>, StorageError>

    pub fn count_in_channel(&self, channel_id: &ChannelId) -> Result<u64, StorageError>

    // Thread operations
    pub fn get_thread(&self, root_message_id: &MessageId) -> Result<Option<Thread>, StorageError>
    pub fn upsert_thread(&self, thread: &Thread) -> Result<(), StorageError>
}
```

For content serialization:
- Use serde_json::to_string for MessageContent → content_json
- Use serde_json::from_str for content_json → MessageContent
- Derive content_type discriminant from MessageContent variant

For pagination in list_for_channel:
```sql
SELECT * FROM messages
WHERE channel_id = ?1
  AND (created_at, id) < (SELECT created_at, id FROM messages WHERE id = ?2)  -- if before is Some
ORDER BY created_at DESC, id DESC
LIMIT ?3
```
If before is None, skip the AND clause.

Add serde_json to Cargo.toml if not present.

AVOID: Don't use OFFSET for pagination (inefficient). Use cursor-based pagination with (created_at, id) tuple comparison. Don't forget to ORDER BY created_at DESC for reverse chronological.
