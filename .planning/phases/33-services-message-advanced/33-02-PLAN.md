//! Draft service.
//!
//! Manage message drafts with auto-save support.

use std::sync::Arc;
use crate::domain::draft::{Draft, DraftKey};
use crate::domain::ids::{AttachmentId, ChannelId, DraftId, MessageId, UserId};
use crate::error::StorageError;
use crate::storage::repositories::draft_repo::DraftRepository;
use crate::storage::repositories::settings_repo::{keys, SettingsRepository};

/// Draft service errors.
#[derive(Debug, thiserror::Error)]
pub enum DraftError {
    /// Operation requires authentication.
    #[error("not logged in")]
    NotLoggedIn,

    /// Draft not found.
    #[error("draft not found")]
    NotFound,

    /// Storage operation failed.
    #[error("storage error: {0}")]
    Storage(#[from] StorageError),
}

/// Draft service for managing message drafts.
pub struct DraftService {
    settings_repo: Arc<SettingsRepository>,
    draft_repo: Arc<DraftRepository>,
}
```

Implement methods:

1. `new(settings_repo, draft_repo) -> Self`

2. `current_user_id(&self) -> Result<UserId, DraftError>` - helper like MessageService

3. `save_draft(&self, channel_id, thread_id: Option<MessageId>, content, reply_to: Option<MessageId>, attachments: Vec<AttachmentId>) -> Result<Draft, DraftError>`
   - Check logged in
   - Build DraftKey from channel_id and thread_id
   - Check if draft exists via get_by_key
   - If exists: update content, reply_to, attachments
   - If not exists: create new Draft
   - Call draft_repo.upsert()
   - Return the draft

4. `get_draft(&self, channel_id, thread_id: Option<MessageId>) -> Result<Option<Draft>, DraftError>`
   - Check logged in
   - Build DraftKey
   - Return draft_repo.get_by_key()

5. `delete_draft(&self, channel_id, thread_id: Option<MessageId>) -> Result<(), DraftError>`
   - Check logged in
   - Build DraftKey
   - Call draft_repo.delete_by_key()
   - Map NotFound to DraftError::NotFound

6. `list_drafts_for_channel(&self, channel_id) -> Result<Vec<Draft>, DraftError>`
   - Check logged in
   - Return draft_repo.list_for_channel()

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_draft_error_display() {
        let err = DraftError::NotLoggedIn;
        assert!(err.to_string().contains("not logged in"));

        let err = DraftError::NotFound;
        assert!(err.to_string().contains("not found"));
    }

    #[test]
    fn test_draft_key_construction() {
        let channel_id = ChannelId::new();
        let thread_id = MessageId::new();

        // Channel-only key
        let key1 = DraftKey::for_channel(channel_id);
        assert!(!key1.is_thread());

        // Thread key
        let key2 = DraftKey::for_thread(channel_id, thread_id);
        assert!(key2.is_thread());
    }
}
```

These tests verify error types and basic key construction without needing database.
