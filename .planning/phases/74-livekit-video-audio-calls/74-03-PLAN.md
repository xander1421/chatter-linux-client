//! LiveKit room connection management.

use livekit::prelude::*;
use livekit::e2ee::{E2eeOptions, EncryptionType, key_provider::{KeyProvider, KeyProviderOptions}};
use tokio::sync::mpsc;
use std::sync::Arc;
use parking_lot::RwLock;

use crate::error::CallError;

/// Current state of the room connection.
#[derive(Clone, Debug, PartialEq)]
pub enum RoomState {
    /// Not connected to any room.
    Disconnected,
    /// Connecting to room.
    Connecting,
    /// Connected and active.
    Connected {
        room_name: String,
        participant_count: usize,
    },
    /// Connection failed.
    Failed { reason: String },
}

/// Room event for UI updates.
#[derive(Clone, Debug)]
pub enum RoomEvent {
    /// Connection state changed.
    StateChanged(RoomState),
    /// Participant joined.
    ParticipantJoined {
        identity: String,
        display_name: Option<String>,
    },
    /// Participant left.
    ParticipantLeft { identity: String },
    /// Track subscribed (remote video/audio available).
    TrackSubscribed {
        participant_identity: String,
        track_sid: String,
        is_video: bool,
    },
    /// Track unsubscribed.
    TrackUnsubscribed {
        participant_identity: String,
        track_sid: String,
    },
    /// E2EE state changed.
    EncryptionStateChanged {
        participant_identity: String,
        is_encrypted: bool,
    },
}

/// Manages LiveKit room connections with E2EE.
pub struct RoomManager {
    /// Current room (if connected).
    room: Arc<RwLock<Option<Room>>>,
    /// Current state.
    state: Arc<RwLock<RoomState>>,
    /// Event sender for UI updates.
    event_tx: mpsc::UnboundedSender<RoomEvent>,
}

impl RoomManager {
    /// Create a new RoomManager.
    pub fn new() -> (Self, mpsc::UnboundedReceiver<RoomEvent>) {
        let (event_tx, event_rx) = mpsc::unbounded_channel();
        let manager = Self {
            room: Arc::new(RwLock::new(None)),
            state: Arc::new(RwLock::new(RoomState::Disconnected)),
            event_tx,
        };
        (manager, event_rx)
    }

    /// Connect to a LiveKit room with E2EE.
    ///
    /// # Arguments
    /// * `url` - LiveKit server URL
    /// * `token` - LiveKit access token (from server)
    /// * `encryption_key` - 32-byte key for E2EE (from Signal derivation)
    pub async fn connect(
        &self,
        url: &str,
        token: &str,
        encryption_key: Vec<u8>,
    ) -> Result<(), CallError> {
        self.set_state(RoomState::Connecting);

        // Create key provider with Signal-derived key
        let key_provider = KeyProvider::with_shared_key(
            KeyProviderOptions::default(),
            encryption_key,
        );

        // Configure room with E2EE
        let room_options = RoomOptions {
            auto_subscribe: true,
            encryption: Some(E2eeOptions {
                key_provider,
                encryption_type: EncryptionType::Gcm,
            }),
            ..Default::default()
        };

        // Connect to room
        let (room, mut events) = Room::connect(url, token, room_options)
            .await
            .map_err(|e| CallError::ConnectionFailed(e.to_string()))?;

        // Enable E2EE
        room.e2ee_manager().set_enabled(true);

        let room_name = room.name().to_string();
        let participant_count = room.remote_participants().len() + 1;

        // Store room
        {
            let mut room_guard = self.room.write();
            *room_guard = Some(room);
        }

        self.set_state(RoomState::Connected {
            room_name,
            participant_count,
        });

        // Spawn event handler
        let event_tx = self.event_tx.clone();
        let state = self.state.clone();
        let room_arc = self.room.clone();

        tokio::spawn(async move {
            while let Some(event) = events.recv().await {
                Self::handle_room_event(event, &event_tx, &state, &room_arc);
            }

            // Room disconnected
            *state.write() = RoomState::Disconnected;
            let _ = event_tx.send(RoomEvent::StateChanged(RoomState::Disconnected));
        });

        Ok(())
    }

    /// Disconnect from the current room.
    pub async fn disconnect(&self) {
        let room = {
            let mut room_guard = self.room.write();
            room_guard.take()
        };

        if let Some(room) = room {
            room.disconnect().await;
        }

        self.set_state(RoomState::Disconnected);
    }

    /// Get current state.
    pub fn state(&self) -> RoomState {
        self.state.read().clone()
    }

    /// Check if connected.
    pub fn is_connected(&self) -> bool {
        matches!(self.state(), RoomState::Connected { .. })
    }

    fn set_state(&self, state: RoomState) {
        *self.state.write() = state.clone();
        let _ = self.event_tx.send(RoomEvent::StateChanged(state));
    }

    fn handle_room_event(
        event: livekit::RoomEvent,
        event_tx: &mpsc::UnboundedSender<RoomEvent>,
        state: &Arc<RwLock<RoomState>>,
        room: &Arc<RwLock<Option<Room>>>,
    ) {
        match event {
            livekit::RoomEvent::ParticipantConnected(participant) => {
                let _ = event_tx.send(RoomEvent::ParticipantJoined {
                    identity: participant.identity().to_string(),
                    display_name: participant.name().map(|s| s.to_string()),
                });

                // Update participant count
                if let Some(room) = room.read().as_ref() {
                    let count = room.remote_participants().len() + 1;
                    if let RoomState::Connected { room_name, .. } = &*state.read() {
                        *state.write() = RoomState::Connected {
                            room_name: room_name.clone(),
                            participant_count: count,
                        };
                    }
                }
            }
            livekit::RoomEvent::ParticipantDisconnected(participant) => {
                let _ = event_tx.send(RoomEvent::ParticipantLeft {
                    identity: participant.identity().to_string(),
                });
            }
            livekit::RoomEvent::TrackSubscribed { track, participant, .. } => {
                let is_video = matches!(track, RemoteTrack::Video(_));
                let _ = event_tx.send(RoomEvent::TrackSubscribed {
                    participant_identity: participant.identity().to_string(),
                    track_sid: track.sid().to_string(),
                    is_video,
                });
            }
            livekit::RoomEvent::TrackUnsubscribed { track, participant, .. } => {
                let _ = event_tx.send(RoomEvent::TrackUnsubscribed {
                    participant_identity: participant.identity().to_string(),
                    track_sid: track.sid().to_string(),
                });
            }
            livekit::RoomEvent::E2eeStateChanged { participant, state: enc_state } => {
                use livekit::webrtc::native::frame_cryptor::EncryptionState;
                let is_encrypted = matches!(enc_state, EncryptionState::Ok);
                let _ = event_tx.send(RoomEvent::EncryptionStateChanged {
                    participant_identity: participant.identity().to_string(),
                    is_encrypted,
                });
            }
            _ => {}
        }
    }
}

//! Call service.
//!
//! Orchestrates call lifecycle: server API + LiveKit + E2EE.

use std::sync::Arc;
use tokio::sync::mpsc;
use parking_lot::RwLock;
use uuid::Uuid;

use crate::calls::{RoomManager, RoomState, LiveKitRoomEvent};
use crate::crypto::call_key_provider::CallKeyState;
use crate::domain::call::{Call, CallType, CallState, Participant};
use crate::domain::ids::{CallId, ChannelId, UserId};
use crate::error::CallError;
use crate::network::grpc::CallClient;
use crate::proto::chat::v1::call as proto;

/// Call service for managing voice/video calls.
pub struct CallService {
    /// gRPC client for server communication.
    call_client: Arc<CallClient>,
    /// LiveKit room manager.
    room_manager: Arc<RoomManager>,
    /// Current active call (if any).
    active_call: Arc<RwLock<Option<ActiveCall>>>,
    /// Room event receiver.
    room_events_rx: Arc<RwLock<Option<mpsc::UnboundedReceiver<LiveKitRoomEvent>>>>,
}

/// State of the currently active call.
struct ActiveCall {
    /// Call domain object.
    call: Call,
    /// E2EE key state.
    key_state: CallKeyState,
    /// LiveKit room name.
    room_name: String,
    /// LiveKit URL.
    livekit_url: String,
}

impl CallService {
    /// Create a new CallService.
    pub fn new(call_client: Arc<CallClient>) -> Self {
        let (room_manager, room_events_rx) = RoomManager::new();
        Self {
            call_client,
            room_manager: Arc::new(room_manager),
            active_call: Arc::new(RwLock::new(None)),
            room_events_rx: Arc::new(RwLock::new(Some(room_events_rx))),
        }
    }

    /// Take the room events receiver (can only be called once).
    pub fn take_room_events(&self) -> Option<mpsc::UnboundedReceiver<LiveKitRoomEvent>> {
        self.room_events_rx.write().take()
    }

    /// Initiate a new call in a channel.
    ///
    /// # Arguments
    /// * `channel_id` - Channel to start call in
    /// * `title` - Call title
    /// * `call_type` - Voice or Video
    /// * `encryption_root_key` - Root key from Signal session for E2EE
    pub async fn initiate_call(
        &self,
        channel_id: ChannelId,
        title: &str,
        call_type: CallType,
        encryption_root_key: &[u8; 32],
    ) -> Result<Call, CallError> {
        // Check if already in a call
        if self.active_call.read().is_some() {
            return Err(CallError::AlreadyInCall);
        }

        // Convert call type to proto
        let proto_type = match call_type {
            CallType::Voice => proto::CallType::Voice,
            CallType::Video => proto::CallType::Video,
            CallType::Group => proto::CallType::Video, // Groups use video capability
        };

        // Create call on server
        let response = self.call_client
            .create_call(channel_id.to_string(), title.to_string(), proto_type, None)
            .await
            .map_err(|e| CallError::ServerError(e.to_string()))?;

        // Derive E2EE key
        let key_state = CallKeyState::new(encryption_root_key, &response.call_id)
            .map_err(|e| CallError::CryptoError(e.to_string()))?;

        // Connect to LiveKit room with encryption
        self.room_manager
            .connect(&response.livekit_url, &response.livekit_token, key_state.current_key_vec())
            .await?;

        // Create domain call object
        let call_id = CallId::parse(&response.call_id)
            .map_err(|_| CallError::InvalidCallId)?;

        let mut call = Call::new(channel_id, UserId::new(), call_type); // TODO: get actual user ID
        call.id = call_id;
        call.start();

        // Store active call
        {
            let mut active = self.active_call.write();
            *active = Some(ActiveCall {
                call: call.clone(),
                key_state,
                room_name: response.room_name,
                livekit_url: response.livekit_url,
            });
        }

        Ok(call)
    }

    /// Join an existing call.
    ///
    /// # Arguments
    /// * `call_id` - Call to join
    /// * `encryption_root_key` - Root key from Signal session for E2EE
    pub async fn join_call(
        &self,
        call_id: CallId,
        encryption_root_key: &[u8; 32],
    ) -> Result<Call, CallError> {
        // Check if already in a call
        if self.active_call.read().is_some() {
            return Err(CallError::AlreadyInCall);
        }

        // Join call on server
        let response = self.call_client
            .join_call(call_id.to_string())
            .await
            .map_err(|e| CallError::ServerError(e.to_string()))?;

        // Derive E2EE key (must match initiator's key)
        let key_state = CallKeyState::new(encryption_root_key, &call_id.to_string())
            .map_err(|e| CallError::CryptoError(e.to_string()))?;

        // Connect to LiveKit room
        self.room_manager
            .connect(&response.livekit_url, &response.livekit_token, key_state.current_key_vec())
            .await?;

        // Convert proto CallInfo to domain Call
        let call = self.proto_call_info_to_domain(&response.call_info)?;

        // Store active call
        {
            let mut active = self.active_call.write();
            *active = Some(ActiveCall {
                call: call.clone(),
                key_state,
                room_name: response.call_info.room_name.clone(),
                livekit_url: response.livekit_url,
            });
        }

        Ok(call)
    }

    /// Leave the current call.
    pub async fn leave_call(&self) -> Result<(), CallError> {
        let call_id = {
            let active = self.active_call.read();
            active.as_ref()
                .map(|a| a.call.id.to_string())
                .ok_or(CallError::NotInCall)?
        };

        // Leave on server
        self.call_client.leave_call(call_id).await
            .map_err(|e| CallError::ServerError(e.to_string()))?;

        // Disconnect from LiveKit
        self.room_manager.disconnect().await;

        // Clear active call
        *self.active_call.write() = None;

        Ok(())
    }

    /// End the current call (creator/admin only).
    pub async fn end_call(&self, reason: Option<&str>) -> Result<(), CallError> {
        let call_id = {
            let active = self.active_call.read();
            active.as_ref()
                .map(|a| a.call.id.to_string())
                .ok_or(CallError::NotInCall)?
        };

        // End on server
        self.call_client.end_call(call_id, reason.map(|s| s.to_string()))
            .await
            .map_err(|e| CallError::ServerError(e.to_string()))?;

        // Disconnect from LiveKit
        self.room_manager.disconnect().await;

        // Clear active call
        *self.active_call.write() = None;

        Ok(())
    }

    /// Get the currently active call.
    pub fn active_call(&self) -> Option<Call> {
        self.active_call.read().as_ref().map(|a| a.call.clone())
    }

    /// Check if currently in a call.
    pub fn is_in_call(&self) -> bool {
        self.active_call.read().is_some()
    }

    /// Get room connection state.
    pub fn room_state(&self) -> RoomState {
        self.room_manager.state()
    }

    fn proto_call_info_to_domain(&self, info: &proto::CallInfo) -> Result<Call, CallError> {
        let call_id = CallId::parse(&info.id)
            .map_err(|_| CallError::InvalidCallId)?;
        let channel_id = ChannelId::parse(&info.channel_id)
            .map_err(|_| CallError::InvalidChannelId)?;
        let initiator_id = UserId::parse(&info.created_by)
            .map_err(|_| CallError::InvalidUserId)?;

        let call_type = match proto::CallType::try_from(info.r#type).unwrap_or(proto::CallType::Unspecified) {
            proto::CallType::Voice => CallType::Voice,
            proto::CallType::Video | proto::CallType::ScreenShare => CallType::Video,
            _ => CallType::Voice,
        };

        let mut call = Call::new(channel_id, initiator_id, call_type);
        call.id = call_id;
        call.start();

        Ok(call)
    }
}
