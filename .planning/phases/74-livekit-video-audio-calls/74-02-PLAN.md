//! Call encryption key derivation.
//!
//! Derives AES-256 keys for LiveKit E2EE from Signal Protocol sessions.
//! This provides Signal-level key establishment with AES-GCM media encryption.

use hkdf::Hkdf;
use sha2::Sha256;
use crate::error::CryptoError;

/// HKDF salt for call key derivation.
const CALL_KEY_SALT: &[u8] = b"ChatterCallKey-v1";

/// Info prefix for call key derivation.
const CALL_KEY_INFO_PREFIX: &[u8] = b"call-encryption:";

/// Derives a 32-byte call encryption key from a session root key.
///
/// For DM calls, the root_key comes from the Signal session.
/// For group calls, the root_key comes from the sender key chain.
///
/// # Arguments
/// * `root_key` - 32-byte root key from Signal session or sender key
/// * `call_id` - Unique call identifier (used as info)
///
/// # Returns
/// 32-byte AES-256 key suitable for LiveKit E2EE.
pub fn derive_call_key(
    root_key: &[u8; 32],
    call_id: &str,
) -> Result<[u8; 32], CryptoError> {
    let hkdf = Hkdf::<Sha256>::new(Some(CALL_KEY_SALT), root_key);

    // Combine prefix with call_id for info parameter
    let mut info = CALL_KEY_INFO_PREFIX.to_vec();
    info.extend_from_slice(call_id.as_bytes());

    let mut call_key = [0u8; 32];
    hkdf.expand(&info, &mut call_key)
        .map_err(|_| CryptoError::KeyDerivationFailed("HKDF expand failed".into()))?;

    Ok(call_key)
}

/// Key provider for LiveKit E2EE integration.
///
/// Wraps the derived key and provides methods for key ratcheting
/// if needed during long calls.
pub struct CallKeyState {
    /// Current encryption key.
    current_key: [u8; 32],
    /// Call ID this key is for.
    call_id: String,
    /// Key epoch (increments on ratchet).
    epoch: u32,
}

impl CallKeyState {
    /// Create a new key state for a call.
    pub fn new(root_key: &[u8; 32], call_id: &str) -> Result<Self, CryptoError> {
        let current_key = derive_call_key(root_key, call_id)?;
        Ok(Self {
            current_key,
            call_id: call_id.to_string(),
            epoch: 0,
        })
    }

    /// Get the current encryption key.
    pub fn current_key(&self) -> &[u8; 32] {
        &self.current_key
    }

    /// Get the current key as a Vec (for LiveKit KeyProvider).
    pub fn current_key_vec(&self) -> Vec<u8> {
        self.current_key.to_vec()
    }

    /// Get the current epoch.
    pub fn epoch(&self) -> u32 {
        self.epoch
    }

    /// Ratchet the key forward (for long-running calls).
    ///
    /// Derives a new key from the current key + epoch.
    pub fn ratchet(&mut self) -> Result<(), CryptoError> {
        self.epoch += 1;

        let info = format!("{}:epoch:{}", self.call_id, self.epoch);
        let hkdf = Hkdf::<Sha256>::new(Some(CALL_KEY_SALT), &self.current_key);

        hkdf.expand(info.as_bytes(), &mut self.current_key)
            .map_err(|_| CryptoError::KeyDerivationFailed("Key ratchet failed".into()))?;

        Ok(())
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_derive_call_key() {
        let root_key = [1u8; 32];
        let call_id = "test-call-123";

        let key1 = derive_call_key(&root_key, call_id).unwrap();
        let key2 = derive_call_key(&root_key, call_id).unwrap();

        // Same inputs = same output
        assert_eq!(key1, key2);

        // Different call_id = different key
        let key3 = derive_call_key(&root_key, "different-call").unwrap();
        assert_ne!(key1, key3);
    }

    #[test]
    fn test_key_state_ratchet() {
        let root_key = [1u8; 32];
        let mut state = CallKeyState::new(&root_key, "test-call").unwrap();

        let initial_key = state.current_key;
        assert_eq!(state.epoch(), 0);

        state.ratchet().unwrap();
        assert_eq!(state.epoch(), 1);
        assert_ne!(*state.current_key(), initial_key);

        // Ratcheting again produces different key
        let after_first = state.current_key;
        state.ratchet().unwrap();
        assert_ne!(*state.current_key(), after_first);
    }
}

Add a method to SignalService that extracts a root key suitable for call key derivation.

For DM sessions, we need to access the session's root key. Since libsignal doesn't expose the root key directly, we'll derive a call-specific key from the identity keys using ECDH + HKDF.

Add this method to SignalService:

```rust
/// Derive a shared secret for call encryption with a peer.
///
/// Uses ECDH between our identity key and peer's identity key,
/// then derives a call-specific key. This is separate from the
/// Double Ratchet session - it's a one-time derivation for calls.
///
/// # Arguments
/// * `peer_address` - The peer's Signal protocol address
/// * `call_id` - Unique call identifier
///
/// # Returns
/// 32-byte key for LiveKit E2EE.
pub async fn derive_call_encryption_key(
    &self,
    peer_address: &ProtocolAddress,
    call_id: &str,
) -> Result<[u8; 32], CryptoError> {
    // Get our identity key
    let our_identity = self.identity_key_store.get_identity_key_pair().await?;

    // Get peer's identity key
    let peer_identity = self.identity_key_store
        .get_identity(&peer_address)
        .await?
        .ok_or_else(|| CryptoError::NoSession)?;

    // Perform ECDH: our private * their public
    // Convert Ed25519 to X25519 for ECDH
    let our_x25519_private = x25519_from_ed25519_private(&our_identity.private_key())?;
    let peer_x25519_public = x25519_from_ed25519_public(&peer_identity.public_key())?;

    // ECDH
    let shared_secret = our_x25519_private.diffie_hellman(&peer_x25519_public);

    // Derive call key
    let root_key: [u8; 32] = shared_secret.as_bytes().clone();
    derive_call_key(&root_key, call_id)
}
```
