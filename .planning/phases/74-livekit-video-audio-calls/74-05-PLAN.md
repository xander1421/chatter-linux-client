                        Text {
                            text: "• " + call-data.participant-count + " participants";
                            font-size: 12px;
                            color: Theme.text-secondary;
                        }
                    }
                }
            }
        }

        // Video grid
        Rectangle {
            vertical-stretch: 1;
            background: Theme.background;

            // Grid layout depends on participant count
            // For 1 participant: full screen
            // For 2: side by side
            // For 3-4: 2x2 grid
            // For 5+: scrollable grid

            if call-data.participant-count == 1: ParticipantTile {
                width: parent.width - 32px;
                height: parent.height - 32px;
                x: 16px;
                y: 16px;
                participant: call-data.participants[0];
                video-frame: local-video-frame;
                is-local: true;
            }

            if call-data.participant-count == 2: HorizontalLayout {
                padding: 16px;
                spacing: 16px;

                ParticipantTile {
                    horizontal-stretch: 1;
                    participant: call-data.participants[0];
                    video-frame: remote-video-frames[0];
                }

                ParticipantTile {
                    horizontal-stretch: 1;
                    participant: call-data.participants[1];
                    video-frame: local-video-frame;
                    is-local: true;
                }
            }

            // TODO: Add grid layouts for 3+ participants
        }

        // Controls bar
        Rectangle {
            height: 88px;
            background: transparent;

            CallControls {
                x: (parent.width - self.width) / 2;
                y: 8px;
                width: 280px;
                is-muted: is-muted;
                is-video-enabled: is-video-enabled;
                is-screen-sharing: is-screen-sharing;
                is-video-call: call-data.is-video-call;

                toggle-mute => { root.toggle-mute(); }
                toggle-video => { root.toggle-video(); }
                toggle-screen-share => { root.toggle-screen-share(); }
                end-call => { root.end-call(); }
            }
        }
    }

    pure function format-duration(seconds: int) -> string {
        // Format as mm:ss or h:mm:ss
        if seconds < 3600 {
            return Math.floor(seconds / 60) + ":" + (Math.mod(seconds, 60) < 10 ? "0" : "") + Math.mod(seconds, 60);
        } else {
            return Math.floor(seconds / 3600) + ":" +
                   (Math.mod(Math.floor(seconds / 60), 60) < 10 ? "0" : "") + Math.mod(Math.floor(seconds / 60), 60) + ":" +
                   (Math.mod(seconds, 60) < 10 ? "0" : "") + Math.mod(seconds, 60);
        }
    }
}

//! Call UI callback wiring.

use std::sync::Arc;
use slint::{ComponentHandle, Weak, SharedString};
use tokio::sync::mpsc;

use crate::app::AppContext;
use crate::services::CallService;
use crate::calls::LiveKitRoomEvent;
use crate::domain::ids::{CallId, ChannelId};

/// Wire all call-related callbacks.
pub fn wire(ui: &crate::MainWindow, ctx: Arc<AppContext>) {
    wire_incoming_call(ui, ctx.clone());
    wire_active_call(ui, ctx.clone());
    wire_call_events(ui, ctx);
}

fn wire_incoming_call(ui: &crate::MainWindow, ctx: Arc<AppContext>) {
    let ui_weak = ui.as_weak();
    let ctx_clone = ctx.clone();

    // Accept call callback
    ui.global::<crate::CallCallbacks>().on_accept_call(move |call_id| {
        let ctx = ctx_clone.clone();
        let ui = ui_weak.clone();
        let call_id_str = call_id.to_string();

        tokio::spawn(async move {
            let call_service = ctx.call_service();

            // Get encryption key from signal service
            // TODO: Get peer address from call data
            let encryption_key = [0u8; 32]; // Placeholder - derive from Signal session

            match call_service.join_call(
                CallId::parse(&call_id_str).unwrap(),
                &encryption_key,
            ).await {
                Ok(call) => {
                    slint::invoke_from_event_loop(move || {
                        if let Some(ui) = ui.upgrade() {
                            ui.set_incoming_call_visible(false);
                            ui.set_active_call_visible(true);
                            // Update call data
                        }
                    }).ok();
                }
                Err(e) => {
                    tracing::error!("Failed to join call: {}", e);
                }
            }
        });
    });

    // Decline call callback
    let ui_weak = ui.as_weak();
    ui.global::<crate::CallCallbacks>().on_decline_call(move |_call_id| {
        if let Some(ui) = ui_weak.upgrade() {
            ui.set_incoming_call_visible(false);
        }
        // TODO: Notify server that call was declined
    });
}

        let ctx = ctx_clone.clone();

            let call_service = ctx.call_service();

            if let Err(e) = call_service.leave_call().await {
                tracing::error!("Failed to leave call: {}", e);
            }

                    ui.set_active_call_visible(false);
                }
            }).ok();

    });
}

fn wire_call_events(ui: &crate::MainWindow, ctx: Arc<AppContext>) {
    let ui_weak = ui.as_weak();
    let call_service = ctx.call_service();

    // Take the room events receiver
    if let Some(mut events) = call_service.take_room_events() {
        tokio::spawn(async move {
            while let Some(event) = events.recv().await {
                let ui = ui_weak.clone();

                match event {
                    LiveKitRoomEvent::StateChanged(state) => {
                        slint::invoke_from_event_loop(move || {
                            if let Some(ui) = ui.upgrade() {
                                // Update connection state in UI
                                tracing::info!("Room state changed: {:?}", state);
                            }
                        }).ok();
                    }
                    LiveKitRoomEvent::ParticipantJoined { identity, display_name } => {
                        slint::invoke_from_event_loop(move || {
                            if let Some(ui) = ui.upgrade() {
                                // Add participant to UI
                                tracing::info!("Participant joined: {}", identity);
                            }
                        }).ok();
                    }
                    LiveKitRoomEvent::ParticipantLeft { identity } => {
                        slint::invoke_from_event_loop(move || {
                            if let Some(ui) = ui.upgrade() {
                                // Remove participant from UI
                                tracing::info!("Participant left: {}", identity);
                            }
                        }).ok();
                    }
                    LiveKitRoomEvent::TrackSubscribed { participant_identity, track_sid, is_video } => {
                        if is_video {
                            // Start video frame streaming for this participant
                            tracing::info!("Video track subscribed: {} from {}", track_sid, participant_identity);
                        }
                    }
                    LiveKitRoomEvent::EncryptionStateChanged { participant_identity, is_encrypted } => {
                        tracing::info!("Encryption state for {}: {}", participant_identity, is_encrypted);
                    }
                    _ => {}
                }
            }
        });
    }
}

/// Start a call from the chat header.
pub async fn start_call(
    ctx: Arc<AppContext>,
    channel_id: ChannelId,
    title: &str,
    is_video: bool,
    peer_user_id: Option<crate::domain::ids::UserId>,
) -> Result<(), Box<dyn std::error::Error + Send + Sync>> {
    let call_service = ctx.call_service();

    // For DM calls, derive key from Signal session
    let encryption_key = if let Some(peer_id) = peer_user_id {
        // TODO: Get key from SignalService.derive_call_encryption_key()
        [0u8; 32] // Placeholder
    } else {
        // For group calls, use a different derivation
        [0u8; 32] // Placeholder
    };

    let call_type = if is_video {
        crate::domain::call::CallType::Video
    } else {
        crate::domain::call::CallType::Voice
    };

    call_service.initiate_call(channel_id, title, call_type, &encryption_key).await?;

    Ok(())
}
