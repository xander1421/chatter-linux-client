<task type="auto">
  <name>Task 2: Create IncomingCallModal component</name>
  <files>ui/calls/incoming_call_modal.slint</files>
  <action>
Create the incoming call modal that appears when receiving a call:

```slint
import { Theme } from "../theme.slint";
import { Avatar } from "../components/avatar.slint";
import { Button } from "../components/button.slint";
import { IncomingCallData } from "types.slint";

export component IncomingCallModal inherits Rectangle {
    in property <bool> visible: false;
    in property <IncomingCallData> call-data;

    callback accept-call();
    callback decline-call();

    // Full screen overlay
    width: 100%;
    height: 100%;
    background: Theme.overlay-background;
    opacity: visible ? 1.0 : 0.0;

    animate opacity { duration: 200ms; }

    if visible: Rectangle {
        // Center card
        x: (parent.width - self.width) / 2;
        y: (parent.height - self.height) / 2;
        width: 320px;
        height: 400px;
        background: Theme.surface;
        border-radius: 16px;
        drop-shadow-blur: 20px;
        drop-shadow-color: #00000066;

        VerticalLayout {
            alignment: center;
            spacing: 24px;
            padding: 32px;

            // Caller avatar with pulse animation
            Rectangle {
                width: 96px;
                height: 96px;
                border-radius: 48px;
                background: Theme.primary.with-alpha(0.2);

                Avatar {
                    width: 80px;
                    height: 80px;
                    x: 8px;
                    y: 8px;
                    image-url: call-data.caller-avatar;
                    fallback-text: call-data.caller-name;
                }

                // Pulse ring animation
                animate background {
                    duration: 1500ms;
                    iteration-count: -1;
                    easing: ease-in-out;
                }
            }

            // Caller info
            VerticalLayout {
                alignment: center;
                spacing: 8px;

                Text {
                    text: call-data.caller-name;
                    font-size: 20px;
                    font-weight: 600;
                    color: Theme.text-primary;
                    horizontal-alignment: center;
                }

                Text {
                    text: call-data.is-video-call ? "Incoming video call..." : "Incoming voice call...";
                    font-size: 14px;
                    color: Theme.text-secondary;
                    horizontal-alignment: center;
                }

                if call-data.channel-name != "": Text {
                    text: "in " + call-data.channel-name;
                    font-size: 12px;
                    color: Theme.text-tertiary;
                    horizontal-alignment: center;
                }
            }

            // Action buttons
            HorizontalLayout {
                alignment: center;
                spacing: 32px;

                // Decline button (red circle)
                Rectangle {
                    width: 64px;
                    height: 64px;
                    border-radius: 32px;
                    background: Theme.error;

                    TouchArea {
                        clicked => { decline-call(); }
                    }

                    Text {
                        text: "✕";
                        font-size: 28px;
                        color: white;
                        horizontal-alignment: center;
                        vertical-alignment: center;
                    }
                }

                // Accept button (green circle)
                Rectangle {
                    width: 64px;
                    height: 64px;
                    border-radius: 32px;
                    background: Theme.success;

                    TouchArea {
                        clicked => { accept-call(); }
                    }

                    Text {
                        text: call-data.is-video-call ? "📹" : "📞";
                        font-size: 28px;
                        horizontal-alignment: center;
                        vertical-alignment: center;
                    }
                }
            }
        }
    }
}
```

  <files>ui/calls/call_controls.slint</files>
  <action>
Create the call control bar with mute, video, screen share, and end call buttons:

```slint
import { Theme } from "../theme.slint";

export component CallControls inherits Rectangle {
    in property <bool> is-muted: false;
    in property <bool> is-video-enabled: true;
    in property <bool> is-screen-sharing: false;
    in property <bool> is-video-call: true;

    callback toggle-mute();
    callback toggle-video();
    callback toggle-screen-share();
    callback end-call();

    height: 72px;
    background: Theme.surface-elevated;
    border-radius: 36px;

    HorizontalLayout {
        alignment: center;
        spacing: 16px;
        padding-left: 24px;
        padding-right: 24px;

        // Mute button
        control-button := Rectangle {
            width: 48px;
            height: 48px;
            border-radius: 24px;
            background: is-muted ? Theme.error : Theme.surface;
            border-width: is-muted ? 0 : 1px;
            border-color: Theme.border;

            TouchArea {
                clicked => { toggle-mute(); }
            }

            Text {
                text: is-muted ? "🔇" : "🎤";
                font-size: 20px;
                horizontal-alignment: center;
                vertical-alignment: center;
            }
        }

        // Video toggle (only for video calls)
        if is-video-call: Rectangle {
            width: 48px;
            height: 48px;
            border-radius: 24px;
            background: !is-video-enabled ? Theme.error : Theme.surface;
            border-width: !is-video-enabled ? 0 : 1px;
            border-color: Theme.border;

            TouchArea {
                clicked => { toggle-video(); }
            }

            Text {
                text: is-video-enabled ? "📹" : "📷";
                font-size: 20px;
                horizontal-alignment: center;
                vertical-alignment: center;
            }
        }

        // Screen share toggle
        Rectangle {
            width: 48px;
            height: 48px;
            border-radius: 24px;
            background: is-screen-sharing ? Theme.primary : Theme.surface;
            border-width: is-screen-sharing ? 0 : 1px;
            border-color: Theme.border;

            TouchArea {
                clicked => { toggle-screen-share(); }
            }

            Text {
                text: "🖥️";
                font-size: 20px;
                horizontal-alignment: center;
                vertical-alignment: center;
            }
        }

        // Spacer
        Rectangle { width: 16px; }

        // End call button (larger, red)
        Rectangle {
            width: 56px;
            height: 48px;
            border-radius: 24px;
            background: Theme.error;

            TouchArea {
                clicked => { end-call(); }
            }

            Text {
                text: "📞";
                font-size: 20px;
                horizontal-alignment: center;
                vertical-alignment: center;
                // Rotated to show "hang up" icon
            }
        }
    }
}
```
