```rust
pub struct MessageItemData {
    pub message_id: String,
    pub sender_id: String,
    pub sender_name: String,
    pub sender_initials: String,
    pub sender_avatar_url: String,
    pub content: String,           // Plain text display
    pub timestamp: String,         // Formatted time (e.g., "2:34 PM", "Yesterday")
    pub is_own: bool,              // Current user's message
    pub is_edited: bool,
    pub is_deleted: bool,
    pub is_system: bool,           // System message (join/leave/etc)
    pub is_ai: bool,               // AI-generated message
    pub has_attachments: bool,
    pub attachment_count: u32,
    pub has_reactions: bool,
    pub reaction_summary: String,  // e.g., "👍 3  ❤️ 2"
    pub reply_count: u32,          // Thread reply count (0 if not thread root)
    pub is_thread_reply: bool,     // Is this a reply in a thread?
    pub show_sender: bool,         // Show sender info (false for consecutive same-sender)
}
```
