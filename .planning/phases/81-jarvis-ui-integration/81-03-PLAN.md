//! Local-only Jarvis messages (not sent to server).

use slint::{ModelRc, VecModel};
use crate::{MainWindow, MessageItemData};

/// Add a local Jarvis response message to the chat view.
///
/// This message is display-only and not sent to the server.
pub fn add_jarvis_response_to_chat(
    window: &MainWindow,
    query: &str,
    response: &str,
) {
    let timestamp = chrono::Local::now().format("%H:%M").to_string();
    let timestamp_ns = chrono::Utc::now().timestamp_nanos_opt().unwrap_or(0) as u64;

    // Create Jarvis response message (local only)
    let msg_item = MessageItemData {
        message_id: format!("jarvis-{}", uuid::Uuid::new_v4()).into(),
        sender_id: "jarvis".into(),
        sender_name: "Jarvis".into(),
        sender_initials: "J".into(),
        sender_avatar_url: "".into(),
        content: format!("**@Jarvis:** {}\n\n{}", query, response).into(),
        timestamp: timestamp.into(),
        timestamp_ns: timestamp_ns.to_string().into(),
        is_own: false,
        is_edited: false,
        is_deleted: false,
        is_system: false,
        is_ai: true,  // Mark as AI message
        has_attachments: false,
        attachment_count: 0,
        attachments: ModelRc::default(),
        has_reactions: false,
        reaction_summary: "".into(),
        reply_count: 0,
        reply_count_text: "".into(),
        is_thread_reply: false,
        show_sender: true,
        delivery_status: "local".into(),  // Mark as local-only
        is_pinned: false,
        is_call: false,
        call_type: "".into(),
        call_duration: "".into(),
    };

    // Add to existing messages
    let model = window.get_messages();
    let mut messages: Vec<_> = (0..model.row_count())
        .filter_map(|i| model.row_data(i))
        .collect();
    messages.push(msg_item);
    messages.sort_by(|a, b| {
        let ts_a = a.timestamp_ns.parse::<u64>().unwrap_or(0);
        let ts_b = b.timestamp_ns.parse::<u64>().unwrap_or(0);
        ts_a.cmp(&ts_b)
    });
    window.set_messages(ModelRc::new(VecModel::from(messages)));
    window.set_message_scroll_position(-99999.0);
    window.set_is_at_bottom(true);
}

### Task 4: Update mention_handler to use local messages
**File:** `src/bin_ui/features/jarvis/mention_handler.rs`

Update `handle_jarvis_mention` to:
1. Add thinking placeholder to chat
2. Replace with real response when done

```rust
pub async fn handle_jarvis_mention(
    ui_ctx: &Arc<UIContext>,
    query: &str,
    channel_id: Option<&ChannelId>,
    weak_window: Weak<MainWindow>,
) {
    // Set thinking state
    ui_ctx.jarvis_state.set_thinking(true);

    // Add thinking placeholder to chat
    let thinking_id = if let Some(window) = weak_window.upgrade() {
        let id = super::local_messages::add_jarvis_thinking_placeholder(&window);
        Some(id)
    } else {
        None
    };

    // Update header thinking indicator
    let weak_header = weak_window.clone();
    let _ = slint::invoke_from_event_loop(move || {
        if let Some(w) = weak_header.upgrade() {
            w.set_jarvis_header_thinking(true);
        }
    });

    // Build context and call Jarvis...
    // (rest of the function remains similar, but use local_messages to show response)

    let context = match channel_id {
        Some(id) => JarvisContext::channel(id.clone()),
        None => JarvisContext::general(),
    };

    let author_id = ui_ctx.auth_state.get().current_user_id;

    match ui_ctx.jarvis_service.ask(&context, query, author_id.as_ref()).await {
        Ok(response) => {
            ui_ctx.jarvis_state.set_thinking(false);
            let response_clone = response.clone();
            let query_clone = query.to_string();
            let thinking_id_clone = thinking_id.clone();

            let _ = slint::invoke_from_event_loop(move || {
                if let Some(w) = weak_window.upgrade() {
                    w.set_jarvis_header_thinking(false);
                    if let Some(tid) = thinking_id_clone {
                        super::local_messages::replace_thinking_with_response(
                            &w,
                            &tid,
                            &query_clone,
                            &response_clone,
                        );
                    }
                }
            });
        }
        Err(e) => {
            ui_ctx.jarvis_state.set_thinking(false);
            let error_msg = format!("Error: {}", e);
            let query_clone = query.to_string();
            let thinking_id_clone = thinking_id.clone();

            let _ = slint::invoke_from_event_loop(move || {
                if let Some(w) = weak_window.upgrade() {
                    w.set_jarvis_header_thinking(false);
                    if let Some(tid) = thinking_id_clone {
                        super::local_messages::replace_thinking_with_response(
                            &w,
                            &tid,
                            &query_clone,
                            &error_msg,
                        );
                    }
                }
            });
        }
    }
}
```
