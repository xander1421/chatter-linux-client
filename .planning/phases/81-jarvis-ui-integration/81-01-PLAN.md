```rust
use std::sync::Arc;
use parking_lot::RwLock;
use uuid::Uuid;

/// Jarvis conversation entry.
#[derive(Debug, Clone)]
pub struct JarvisMessage {
    pub id: String,
    pub role: JarvisRole,
    pub content: String,
    pub timestamp: chrono::DateTime<chrono::Utc>,
    pub context_channel_id: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum JarvisRole {
    User,
    Assistant,
}

#[derive(Debug, Clone, Default)]
struct JarvisStateData {
    /// Whether Jarvis panel is visible.
    panel_visible: bool,
    /// Whether Jarvis is currently thinking (inference in progress).
    is_thinking: bool,
    /// Conversation history for Jarvis panel.
    messages: Vec<JarvisMessage>,
    /// Current channel context for @Jarvis mentions.
    active_context_channel_id: Option<String>,
}

#[derive(Clone)]
pub struct JarvisState {
    inner: Arc<RwLock<JarvisStateData>>,
}

impl JarvisState {
    pub fn new() -> Self {
        Self {
            inner: Arc::new(RwLock::new(JarvisStateData::default())),
        }
    }

    pub fn is_panel_visible(&self) -> bool {
        self.inner.read().panel_visible
    }

    pub fn toggle_panel(&self) {
        let mut state = self.inner.write();
        state.panel_visible = !state.panel_visible;
    }

    pub fn show_panel(&self) {
        self.inner.write().panel_visible = true;
    }

    pub fn hide_panel(&self) {
        self.inner.write().panel_visible = false;
    }

    pub fn is_thinking(&self) -> bool {
        self.inner.read().is_thinking
    }

    pub fn set_thinking(&self, thinking: bool) {
        self.inner.write().is_thinking = thinking;
    }

    pub fn get_messages(&self) -> Vec<JarvisMessage> {
        self.inner.read().messages.clone()
    }

    pub fn add_user_message(&self, content: &str, channel_id: Option<&str>) -> String {
        let id = Uuid::new_v4().to_string();
        let mut state = self.inner.write();
        state.messages.push(JarvisMessage {
            id: id.clone(),
            role: JarvisRole::User,
            content: content.to_string(),
            timestamp: chrono::Utc::now(),
            context_channel_id: channel_id.map(|s| s.to_string()),
        });
        id
    }

    pub fn add_assistant_message(&self, content: &str) -> String {
        let id = Uuid::new_v4().to_string();
        let mut state = self.inner.write();
        state.messages.push(JarvisMessage {
            id: id.clone(),
            role: JarvisRole::Assistant,
            content: content.to_string(),
            timestamp: chrono::Utc::now(),
            context_channel_id: None,
        });
        id
    }

    pub fn set_active_context(&self, channel_id: Option<&str>) {
        self.inner.write().active_context_channel_id = channel_id.map(|s| s.to_string());
    }

    pub fn get_active_context(&self) -> Option<String> {
        self.inner.read().active_context_channel_id.clone()
    }

    pub fn clear_messages(&self) {
        self.inner.write().messages.clear();
    }
}

impl Default for JarvisState {
    fn default() -> Self {
        Self::new()
    }
}
```

//! Handle @Jarvis mentions in messages.

use std::sync::Arc;
use chatter_linux::app::UIContext;
use chatter_linux::domain::jarvis::JarvisContext;
use chatter_linux::domain::ids::ChannelId;
use slint::Weak;
use crate::MainWindow;

/// Check if message contains @Jarvis mention.
pub fn contains_jarvis_mention(text: &str) -> bool {
    let lower = text.to_lowercase();
    lower.contains("@jarvis")
}

/// Extract the query portion after @Jarvis.
pub fn extract_jarvis_query(text: &str) -> Option<String> {
    let lower = text.to_lowercase();
    if let Some(pos) = lower.find("@jarvis") {
        let after = &text[pos + 7..]; // "@jarvis" is 7 chars
        let query = after.trim();
        if !query.is_empty() {
            return Some(query.to_string());
        }
    }
    None
}

/// Handle a @Jarvis mention asynchronously.
pub async fn handle_jarvis_mention(
    ui_ctx: &Arc<UIContext>,
    query: &str,
    channel_id: Option<&ChannelId>,
    weak_window: Weak<MainWindow>,
) {
    // Set thinking state
    ui_ctx.jarvis_state.set_thinking(true);

    // Build context
    let context = match channel_id {
        Some(id) => JarvisContext::channel(id.clone()),
        None => JarvisContext::general(),
    };

    // Get current user ID
    let author_id = ui_ctx.auth_state.get().current_user_id;

    // Add user message to Jarvis state
    let channel_id_str = channel_id.map(|id| id.to_string());
    ui_ctx.jarvis_state.add_user_message(query, channel_id_str.as_deref());

    // Call Jarvis
    match ui_ctx.jarvis_service.ask(&context, query, author_id.as_ref()).await {
        Ok(response) => {
            // Add assistant response to state
            ui_ctx.jarvis_state.add_assistant_message(&response);

            // Update UI via invoke_from_event_loop
            let response_clone = response.clone();
            let jarvis_state = ui_ctx.jarvis_state.clone();
            let _ = slint::invoke_from_event_loop(move || {
                jarvis_state.set_thinking(false);
                if let Some(window) = weak_window.upgrade() {
                    // Show Jarvis panel with response
                    window.set_jarvis_panel_visible(true);
                    window.set_jarvis_response(response_clone.into());
                    window.set_jarvis_thinking(false);
                }
            });
        }
        Err(e) => {
            tracing::error!("Jarvis error: {}", e);
            let error_msg = format!("I encountered an error: {}", e);
            ui_ctx.jarvis_state.add_assistant_message(&error_msg);

            let jarvis_state = ui_ctx.jarvis_state.clone();
            let _ = slint::invoke_from_event_loop(move || {
                jarvis_state.set_thinking(false);
                if let Some(window) = weak_window.upgrade() {
                    window.set_jarvis_panel_visible(true);
                    window.set_jarvis_response(error_msg.into());
                    window.set_jarvis_thinking(false);
                }
            });
        }
    }
}

//! Jarvis slash command parsing.

/// Jarvis slash commands.
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum JarvisCommand {
    /// /ask <query> - Ask Jarvis a question
    Ask(String),
    /// /summarize - Summarize current conversation
    Summarize,
    /// /knowledge <entity> - Get knowledge about an entity
    Knowledge(String),
    /// /remember <fact> - Store a fact in memory
    Remember(String),
}

/// Parse a slash command from message text.
/// Returns None if not a recognized Jarvis command.
pub fn parse_slash_command(text: &str) -> Option<JarvisCommand> {
    let text = text.trim();

    if let Some(rest) = text.strip_prefix("/ask ") {
        let query = rest.trim();
        if !query.is_empty() {
            return Some(JarvisCommand::Ask(query.to_string()));
        }
    }

    if text == "/summarize" || text.starts_with("/summarize ") {
        return Some(JarvisCommand::Summarize);
    }

    if let Some(rest) = text.strip_prefix("/knowledge ") {
        let entity = rest.trim();
        if !entity.is_empty() {
            return Some(JarvisCommand::Knowledge(entity.to_string()));
        }
    }

    if let Some(rest) = text.strip_prefix("/remember ") {
        let fact = rest.trim();
        if !fact.is_empty() {
            return Some(JarvisCommand::Remember(fact.to_string()));
        }
    }

    None
}

// Check for @Jarvis mention
if crate::bin_ui::features::jarvis::mention_handler::contains_jarvis_mention(&text) {
