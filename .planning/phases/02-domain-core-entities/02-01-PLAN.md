```rust
use serde::{Deserialize, Serialize};
use uuid::Uuid;
use std::fmt;
use std::str::FromStr;

/// Unique identifier for a user (UUID v4).
#[derive(Clone, Copy, PartialEq, Eq, Hash, Debug, Serialize, Deserialize)]
pub struct UserId(pub Uuid);

/// Unique identifier for a channel (UUID v4).
#[derive(Clone, Copy, PartialEq, Eq, Hash, Debug, Serialize, Deserialize)]
pub struct ChannelId(pub Uuid);

/// Unique identifier for a message (UUID v4 or ULID).
#[derive(Clone, Copy, PartialEq, Eq, Hash, Debug, Serialize, Deserialize)]
pub struct MessageId(pub Uuid);

/// Unique identifier for a device (u32, matches Signal protocol).
#[derive(Clone, Copy, PartialEq, Eq, Hash, Debug, Serialize, Deserialize)]
pub struct DeviceId(pub u32);

/// Unique identifier for an attachment (UUID v4).
#[derive(Clone, Copy, PartialEq, Eq, Hash, Debug, Serialize, Deserialize)]
pub struct AttachmentId(pub Uuid);

/// Unique identifier for a call (UUID v4).
#[derive(Clone, Copy, PartialEq, Eq, Hash, Debug, Serialize, Deserialize)]
pub struct CallId(pub Uuid);

/// Unique identifier for a session (UUID v4).
#[derive(Clone, Copy, PartialEq, Eq, Hash, Debug, Serialize, Deserialize)]
pub struct SessionId(pub Uuid);

/// Unique identifier for a reaction (UUID v4).
#[derive(Clone, Copy, PartialEq, Eq, Hash, Debug, Serialize, Deserialize)]
pub struct ReactionId(pub Uuid);

/// Unique identifier for an AI integration (UUID v4).
#[derive(Clone, Copy, PartialEq, Eq, Hash, Debug, Serialize, Deserialize)]
pub struct IntegrationId(pub Uuid);

/// Unique identifier for a thread (same as MessageId of root message).
pub type ThreadId = MessageId;
```

Implement for each UUID-based ID type:
- `new()` - generate new random ID
- `from_uuid(uuid: Uuid)` - wrap existing UUID
- `as_uuid(&self)` - get inner UUID reference
- `Display` trait - format as string
- `FromStr` trait - parse from string

For DeviceId, implement:
- `new(id: u32)` - wrap existing u32
- `as_u32(&self)` - get inner value
- `Display` trait

DO NOT implement `Default` for IDs - IDs should always be explicitly created.

```rust
use crate::domain::ids::{DeviceId, UserId};
use chrono::{DateTime, Utc};
use serde::{Deserialize, Serialize};

/// User presence status.
#[derive(Clone, Copy, Debug, PartialEq, Eq, Default, Serialize, Deserialize)]
pub enum PresenceStatus {
    #[default]
    Offline,
    Online,
    Away,
    DoNotDisturb,
}

/// Represents a user in the system.
#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
pub struct User {
    pub id: UserId,
    pub username: String,
    /// Ed25519 public key for identity verification
    pub public_key: Vec<u8>,
    pub last_active: DateTime<Utc>,
    pub created_at: DateTime<Utc>,
}

/// User profile information (display data).
#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
pub struct Profile {
    pub user_id: UserId,
    pub display_name: String,
    pub avatar_url: Option<String>,
    pub status_message: Option<StatusMessage>,
    pub presence: PresenceStatus,
}

/// Custom status message (mood text).
#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
pub struct StatusMessage {
    pub text: String,
    pub emoji: Option<String>,
    pub expires_at: Option<DateTime<Utc>>,
}

/// User's device for multi-device support.
#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
pub struct Device {
    pub id: DeviceId,
    pub user_id: UserId,
    pub name: String,
    pub registration_id: u32,
    pub created_at: DateTime<Utc>,
    pub last_seen: DateTime<Utc>,
}

/// Friend relationship between users.
#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
pub struct Friendship {
    pub user_id: UserId,
    pub friend_id: UserId,
    pub status: FriendshipStatus,
    pub created_at: DateTime<Utc>,
}

/// Friend request status.
#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
pub enum FriendshipStatus {
    Pending,
    Accepted,
    Blocked,
}
```

```rust
use crate::domain::ids::{AttachmentId, ChannelId, IntegrationId, MessageId, UserId};
use chrono::{DateTime, Duration, Utc};
use serde::{Deserialize, Serialize};

/// Represents a chat message.
#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
pub struct Message {
    pub id: MessageId,
    pub channel_id: ChannelId,
    /// None for sealed sender messages (server doesn't know sender)
    pub sender_id: Option<UserId>,
    pub content: MessageContent,
    /// Parent message ID for threads
    pub thread_id: Option<MessageId>,
    /// Message being replied to (quote)
    pub reply_to: Option<MessageId>,
    pub attachments: Vec<AttachmentId>,
    pub is_pinned: bool,
    pub is_edited: bool,
    pub is_deleted: bool,
    pub created_at: DateTime<Utc>,
    pub updated_at: DateTime<Utc>,
    pub edited_at: Option<DateTime<Utc>>,
    /// Reaction counts by emoji
    pub reaction_counts: Vec<ReactionCount>,
    /// Current user's reactions (for UI state)
    pub current_user_reactions: Vec<String>,
    /// AI integrations mentioned in message
    pub mentioned_integrations: Vec<IntegrationId>,
    pub targets_ai_bot: bool,
}

/// Reaction count for a specific emoji.
#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
pub struct ReactionCount {
    pub emoji: String,
    pub count: u32,
}

/// Message content types.
#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
pub enum MessageContent {
    /// Plain text message
    Text(String),
    /// Signal Protocol encrypted content
    Encrypted(EncryptedContent),
    /// Sealed sender (encrypted + anonymous)
    SealedSender(SealedSenderContent),
    /// System message (join, leave, etc)
    System(SystemMessage),
    /// Call notification (started, ended)
    Call(CallMessage),
    /// AI-generated message
    AI(AIMessage),
    /// Mixed text + files
    Mixed(MixedContent),
}

/// Encrypted message content (Signal Protocol).
#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
pub struct EncryptedContent {
    pub ciphertext: Vec<u8>,
    /// Sender key ID for group messages
    pub sender_key_id: Option<u32>,
    pub message_type: EncryptedMessageType,
}

/// Encrypted message type variants.
#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
pub enum EncryptedMessageType {
    /// Standard Signal protocol (PreKey or Whisper message)
    Signal,
    /// Sender key for groups
    SenderKey,
}

/// Sealed sender message (hides sender from server).
#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
pub struct SealedSenderContent {
    pub ciphertext: Vec<u8>,
}

/// System message types.
#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
pub enum SystemMessage {
    UserJoined(UserId),
    UserLeft(UserId),
    ChannelCreated,
    ChannelRenamed { old_name: String, new_name: String },
    UserAdded { by: UserId, added: UserId },
    UserRemoved { by: UserId, removed: UserId },
}

/// Call message for call notifications in chat.
#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
pub struct CallMessage {
    pub call_id: String,
    pub call_type: CallMessageType,
    pub initiator_id: UserId,
    pub initiator_name: String,
    pub started_at: Option<DateTime<Utc>>,
    pub ended_at: Option<DateTime<Utc>>,
    pub duration: Option<Duration>,
    pub participant_count: u32,
}

/// Call message type.
#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
pub enum CallMessageType {
    Started,
    Ended,
    Missed,
}

/// AI-generated message content.
#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
pub struct AIMessage {
    pub text: String,
    pub integration_id: IntegrationId,
    pub model: Option<String>,
    pub prompt_tokens: Option<u32>,
    pub completion_tokens: Option<u32>,
}

/// Mixed content (text + files).
#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
pub struct MixedContent {
    pub text: Option<String>,
    pub attachment_ids: Vec<AttachmentId>,
}

/// Message thread (conversation branch).
#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
pub struct Thread {
    pub root_message_id: MessageId,
    pub channel_id: ChannelId,
    pub reply_count: u32,
    pub last_reply_at: Option<DateTime<Utc>>,
    pub participant_ids: Vec<UserId>,
}
```

Implement for Message:
- `new_text(channel_id, sender_id, text)` - create text message with timestamps
- `is_thread_root(&self)` - true if thread_id is None and has replies
- `is_thread_reply(&self)` - true if thread_id is Some
- `as_text(&self)` - extract text content if Text/Mixed variant
