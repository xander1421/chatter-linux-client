use std::sync::Arc;
use parking_lot::RwLock;
use tonic::service::Interceptor;
use tonic::{Request, Status};

/// Authentication modes for gRPC requests
#[derive(Clone)]
pub enum AuthMode {
    /// Bearer token (device token)
    Bearer(String),
    /// Basic auth (Signal-style: {uuid}.{device_id}:token)
    Basic { username: String, password: String },
    /// Sealed sender with UAK (no auth header, just UAK)
    SealedSender { access_key: [u8; 16] },
    /// No authentication
    None,
}

/// Thread-safe auth token storage
#[derive(Clone)]
pub struct AuthInterceptor {
    auth: Arc<RwLock<AuthMode>>,
}

impl AuthInterceptor {
    pub fn new() -> Self;
    pub fn set_bearer(&self, token: String);
    pub fn set_basic(&self, username: String, password: String);
    pub fn set_sealed_sender(&self, access_key: [u8; 16]);
    pub fn clear(&self);
}

impl Interceptor for AuthInterceptor {
    fn call(&mut self, mut request: Request<()>) -> Result<Request<()>, Status> {
        // Read auth mode and inject appropriate header
        // Bearer -> "authorization: Bearer {token}"
        // Basic -> "authorization: Basic base64({username}:{password})"
        // SealedSender -> "unidentified-access-key: base64({key})"
        // None -> no header
    }
}
```

Use `base64::engine::general_purpose::STANDARD` for encoding.

Do NOT use async - Interceptor::call is sync. Use parking_lot::RwLock (not tokio::sync::RwLock).

Remove RetryInterceptor and LoggingInterceptor stubs - we'll add tower middleware later.

use std::sync::Arc;
use parking_lot::RwLock;
use tokio::sync::broadcast;

/// Connection state
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ConnectionState {
    Disconnected,
    Connecting,
    Connected,
    Reconnecting { attempt: u32 },
    Failed { reason: FailReason },
}

/// Failure reasons for terminal states
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum FailReason {
    AuthExpired,
    DeviceDelinked,
    MaxRetriesExceeded,
}

/// Connection events for subscribers
#[derive(Clone, Debug)]
pub enum ConnectionEvent {
    StateChanged { old: ConnectionState, new: ConnectionState },
    Online,
    Offline,
    AuthRequired,
}

/// State machine for connection lifecycle
pub struct ConnectionStateMachine {
    state: Arc<RwLock<ConnectionState>>,
    events: broadcast::Sender<ConnectionEvent>,
}

impl ConnectionStateMachine {
    pub fn new() -> Self;

    /// Get current state
    pub fn state(&self) -> ConnectionState;

    /// Subscribe to connection events
    pub fn subscribe(&self) -> broadcast::Receiver<ConnectionEvent>;

    /// Transition to new state (emits events)
    pub fn transition(&self, new_state: ConnectionState);

    /// Mark as connecting
    pub fn connecting(&self);

    /// Mark as connected
    pub fn connected(&self);

    /// Mark as disconnected
    pub fn disconnected(&self);

    /// Mark as reconnecting with attempt count
    pub fn reconnecting(&self, attempt: u32);

    /// Mark as failed with reason
    pub fn failed(&self, reason: FailReason);
}
```

Emit appropriate ConnectionEvent on each transition:
- Disconnected → Connecting: StateChanged
- Connecting → Connected: StateChanged + Online
- Connected → Disconnected: StateChanged + Offline
- Connected → Reconnecting: StateChanged + Offline
- Reconnecting → Connected: StateChanged + Online
- Any → Failed(AuthExpired): StateChanged + AuthRequired

Broadcast channel capacity: 16 (events are small, consumers should keep up).

Add unit tests:
- `test_initial_state` - Starts disconnected
- `test_transition_emits_events` - Verify events fired
- `test_online_offline_events` - Connected triggers Online, Disconnected triggers Offline
