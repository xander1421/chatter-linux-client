```rust
use std::time::Duration;
use rand::Rng;

pub struct FibonacciBackoff {
    timeouts: &'static [Duration],
    index: usize,
    jitter: Duration,
}

impl FibonacciBackoff {
    /// Standard timeouts (1s to 55s)
    pub const STANDARD: &'static [Duration] = &[
        Duration::from_secs(1),
        Duration::from_secs(2),
        Duration::from_secs(3),
        Duration::from_secs(5),
        Duration::from_secs(8),
        Duration::from_secs(13),
        Duration::from_secs(21),
        Duration::from_secs(34),
        Duration::from_secs(55),
    ];

    /// Extended timeouts for prolonged offline (up to ~10 min)
    pub const EXTENDED: &'static [Duration] = &[
        // STANDARD timeouts first...
        Duration::from_secs(1),
        Duration::from_secs(2),
        Duration::from_secs(3),
        Duration::from_secs(5),
        Duration::from_secs(8),
        Duration::from_secs(13),
        Duration::from_secs(21),
        Duration::from_secs(34),
        Duration::from_secs(55),
        // Extended...
        Duration::from_secs(89),
        Duration::from_secs(144),
        Duration::from_secs(233),
        Duration::from_secs(377),
        Duration::from_secs(610),
    ];

    pub fn new(timeouts: &'static [Duration]) -> Self;
    pub fn with_jitter(timeouts: &'static [Duration], jitter: Duration) -> Self;

    /// Get next timeout and advance index
    pub fn next(&mut self) -> Duration;

    /// Reset to beginning with optional new timeout sequence
    pub fn reset(&mut self);
    pub fn reset_with(&mut self, timeouts: &'static [Duration]);
}
```

Default jitter: 5 seconds. Use `rand::thread_rng().gen_range(Duration::ZERO..self.jitter)`.

Add unit tests:
- `test_fibonacci_sequence` - Verify timeouts follow Fibonacci
- `test_jitter_bounds` - Verify jitter within range
- `test_reset` - Verify index resets
- `test_extended_sequence` - Verify extended timeouts

Do NOT use async in this module - FibonacciBackoff is sync.
