   -- Server blind signature public keys (for key rotation)
   CREATE TABLE mailbox_server_keys (
       key_id TEXT PRIMARY KEY,
       public_key_der BLOB NOT NULL,
       valid_until INTEGER NOT NULL,  -- Unix timestamp ms
       created_at TEXT NOT NULL
   );

   -- Pending blinding operations (awaiting server signature)
   CREATE TABLE mailbox_pending_blinds (
       id TEXT PRIMARY KEY,  -- UUID
       ephemeral_public_key BLOB NOT NULL,  -- Ed25519 32 bytes
       ephemeral_private_key BLOB NOT NULL,  -- Ed25519 64 bytes (for later signing)
       blinded_message BLOB NOT NULL,
       blinding_secret BLOB NOT NULL,  -- Serialized BlindingSecret
       server_key_id TEXT NOT NULL,
       created_at TEXT NOT NULL,
       FOREIGN KEY (server_key_id) REFERENCES mailbox_server_keys(key_id)
   );

   -- Ready-to-use tokens
   CREATE TABLE mailbox_tokens (
       id TEXT PRIMARY KEY,  -- UUID
       ephemeral_public_key BLOB NOT NULL UNIQUE,  -- Ed25519 32 bytes
       ephemeral_private_key BLOB NOT NULL,  -- Ed25519 64 bytes
       signature BLOB NOT NULL,  -- Unblinded RSA signature
       server_key_id TEXT NOT NULL,
       created_at TEXT NOT NULL,
       expires_at TEXT,  -- Optional expiry
       used_at TEXT,  -- NULL if unused, timestamp if spent
       FOREIGN KEY (server_key_id) REFERENCES mailbox_server_keys(key_id)
   );
   CREATE INDEX idx_mailbox_tokens_unused ON mailbox_tokens(used_at) WHERE used_at IS NULL;

   -- Quota tracking
   CREATE TABLE mailbox_quota (
       id INTEGER PRIMARY KEY CHECK (id = 1),  -- Single row
       tokens_used_today INTEGER NOT NULL DEFAULT 0,
       quota_resets_at INTEGER NOT NULL,  -- Unix timestamp ms
       last_updated TEXT NOT NULL

1. Replace src/mailbox/token_manager.rs stub with full implementation:

```rust
//! Token manager for blind-signed mailbox tokens.
//!
//! Manages the full lifecycle of Chaum blind signature tokens:
//! - Fetching server's RSA public key
//! - Generating ephemeral Ed25519 keypairs
//! - Blinding keys for server signing
//! - Unblinding signatures to create usable tokens
//! - Storing tokens securely in SQLCipher
//! - Tracking daily quota
```

Imports:
- ed25519_dalek::{SigningKey, VerifyingKey} for ephemeral keypairs
- rand::rngs::OsRng for secure key generation
- crate::mailbox::blind_signature::{ServerPublicKey, BlindingOperation, UnblindedToken, blind_key, unblind}
- crate::network::grpc::MailboxClient
- crate::storage::Database
- uuid::Uuid

Struct `TokenManager`:
```rust
pub struct TokenManager {
    db: Arc<Database>,
    // Note: MailboxClient passed to methods, not stored (async lifetime issues)
}
```

Methods:

- `new(db: Arc<Database>) -> Self`

- `async fn refresh_server_key(&self, client: &mut MailboxClient) -> Result<ServerPublicKey>`
  Calls client.get_blind_signature_public_key(), parses DER, stores in mailbox_server_keys table
  Returns cached key if still valid (valid_until > now)

- `fn get_cached_server_key(&self) -> Result<Option<ServerPublicKey>>`
  Query most recent valid key from mailbox_server_keys

- `async fn request_tokens(&self, client: &mut MailboxClient, count: u32) -> Result<u32>`
  1. Check quota (fail if exhausted)
  2. Get/refresh server key
  3. Generate `count` ephemeral Ed25519 keypairs
  4. Blind each public key using blind_key()
  5. Store pending blinds in mailbox_pending_blinds
  6. Call client.get_mailbox_tokens(count, blinded_keys)
  7. For each blinded_signature: unblind and store in mailbox_tokens
  8. Delete pending blinds
  9. Update quota from response
  10. Return number of tokens acquired

- `fn get_available_token(&self) -> Result<Option<MailboxToken>>`
  Query one unused token (used_at IS NULL), return with private key for signing

- `fn mark_token_used(&self, token_id: &str) -> Result<()>`
  UPDATE mailbox_tokens SET used_at = now WHERE id = ?

- `fn available_token_count(&self) -> Result<u32>`
  COUNT(*) FROM mailbox_tokens WHERE used_at IS NULL

- `fn get_quota_status(&self) -> Result<QuotaStatus>`
  Returns QuotaStatus { remaining: u32, resets_at: u64 }

- `fn reset_quota_if_needed(&self) -> Result<()>`
  If current time > quota_resets_at, reset tokens_used_today to 0

Types:
- `MailboxToken`: Contains token_id, ephemeral_public_key, ephemeral_private_key, signature
- `QuotaStatus`: { remaining: u32, resets_at: u64 }

Constants:
- `DEFAULT_TOKEN_REQUEST_COUNT: u32 = 5` - How many to request at once
- `MIN_TOKEN_THRESHOLD: u32 = 2` - Request more when below this
