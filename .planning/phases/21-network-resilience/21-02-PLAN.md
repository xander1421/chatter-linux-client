use std::time::{Duration, Instant};
use uuid::Uuid;
use serde::{Serialize, Deserialize};

/// Types of operations that can be queued.
#[derive(Clone, Debug, Serialize, Deserialize)]
pub enum OperationType {
    /// Send a message to a channel
    SendMessage {
        channel_id: String,
        content: Vec<u8>,  // Serialized message content
    },
    /// Send a reaction
    SendReaction {
        message_id: String,
        emoji: String,
    },
    /// Mark messages as read
    MarkRead {
        channel_id: String,
        up_to_message_id: String,
    },
    /// Send typing indicator
    SendTyping {
        channel_id: String,
    },
    /// Edit a message
    EditMessage {
        message_id: String,
        content: Vec<u8>,
    },
    /// Delete a message
    DeleteMessage {
        message_id: String,
    },
}

/// A queued operation waiting for network connectivity.
#[derive(Clone, Debug, Serialize, Deserialize)]
pub struct QueuedOperation {
    /// Unique identifier for this operation
    pub id: Uuid,
    /// The operation to perform
    pub operation: OperationType,
    /// When the operation was queued
    pub queued_at: u64,  // Unix timestamp millis
    /// Number of replay attempts
    pub attempts: u32,
    /// Maximum attempts before dropping (None = unlimited)
    pub max_attempts: Option<u32>,
    /// Priority (lower = higher priority)
    pub priority: u8,
}

/// Queue for offline operations.
pub struct OfflineQueue {
    /// In-memory queue (sorted by priority, then queued_at)
    operations: parking_lot::RwLock<Vec<QueuedOperation>>,
    /// Maximum queue size
    max_size: usize,
    /// Event sender for queue changes
    events: tokio::sync::broadcast::Sender<QueueEvent>,
}

#[derive(Clone, Debug)]
pub enum QueueEvent {
    /// Operation added to queue
    Added(Uuid),
    /// Operation removed (completed or expired)
    Removed(Uuid),
    /// Queue cleared
    Cleared,
    /// Replay started
    ReplayStarted { count: usize },
    /// Replay completed
    ReplayCompleted { succeeded: usize, failed: usize },
}
```

Implement:
- `OfflineQueue::new(max_size: usize)` - creates queue with capacity limit
- `enqueue(&self, op: OperationType) -> Result<Uuid, QueueError>` - adds operation, returns ID
- `dequeue(&self) -> Option<QueuedOperation>` - removes and returns highest priority op
- `peek(&self) -> Option<QueuedOperation>` - view next op without removing
- `remove(&self, id: Uuid) -> bool` - remove specific operation
- `clear(&self)` - remove all operations
- `len(&self) -> usize` and `is_empty(&self) -> bool`
- `subscribe(&self) -> broadcast::Receiver<QueueEvent>`

QueueError enum:
- `QueueFull` - queue at capacity
- `OperationTooLarge` - serialized op exceeds size limit

impl OfflineQueue {
    /// Save queue to persistent storage.
    ///
    /// Serializes queue to JSON and writes to the given path.
    /// Called periodically and on app shutdown.
    pub async fn save(&self, path: &std::path::Path) -> std::io::Result<()>;

    /// Load queue from persistent storage.
    ///
    /// Reads and deserializes queue from the given path.
    /// Called on app startup.
    pub async fn load(&self, path: &std::path::Path) -> std::io::Result<usize>;

    /// Get all operations ready for replay.
    ///
    /// Returns operations sorted by priority and queue time.
    /// Does not remove them from the queue.
    pub fn pending_operations(&self) -> Vec<QueuedOperation>;

    /// Mark an operation as attempted.
    ///
    /// Increments attempt count. Returns false if max attempts exceeded
    /// (operation should be removed).
    pub fn mark_attempted(&self, id: Uuid) -> bool;

    /// Get queue statistics.
    pub fn stats(&self) -> QueueStats;
}

pub struct QueueStats {
    pub total_operations: usize,
    pub oldest_operation_age: Option<Duration>,
    pub operations_by_type: std::collections::HashMap<String, usize>,
}
```

Implementation notes:
- Use serde_json for serialization
- save() should use tokio::fs::write with atomic rename pattern (write to .tmp, rename)
- load() should handle missing file gracefully (return Ok(0))
- pending_operations() returns a clone of the Vec, sorted
- mark_attempted() returns false when attempts >= max_attempts (caller should remove)
- For operations_by_type, use std::any::type_name or match on OperationType variant name
