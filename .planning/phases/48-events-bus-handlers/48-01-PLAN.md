```rust
/// UI-specific events for view layer reactivity.
///
/// These events trigger UI updates that don't correspond to domain changes.
/// Domain events (message received, typing, etc.) are separate.
#[derive(Debug, Clone)]
pub enum UiEvent {
    // Navigation events
    ChannelSelected { channel_id: ChannelId },
    ThreadOpened { channel_id: ChannelId, thread_id: MessageId },
    ThreadClosed { thread_id: MessageId },

    // Modal events
    ModalOpened { modal: ModalType },
    ModalClosed { modal: ModalType },

    // Scroll events (for virtualization)
    ScrollToMessage { message_id: MessageId },
    ScrollToBottom { channel_id: ChannelId },

    // Input events
    DraftUpdated { channel_id: ChannelId, thread_id: Option<MessageId> },
    ComposerFocused { channel_id: ChannelId },
    ComposerBlurred { channel_id: ChannelId },

    // Sidebar events
    SidebarToggled { expanded: bool },
    SidebarSectionToggled { section: SidebarSection, expanded: bool },

    // Search events
    SearchOpened,
    SearchClosed,
    SearchQueryChanged { query: String },

    // Selection events
    MessageSelected { message_id: MessageId },
    MessageDeselected { message_id: MessageId },
    SelectionCleared,

    // Settings events
    SettingsOpened { tab: SettingsTab },
    SettingsClosed,

    // Toast notifications (transient UI feedback)
    ToastShown { id: String, message: String, level: ToastLevel },
    ToastDismissed { id: String },
}

/// Sidebar sections for collapse state.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SidebarSection {
    Channels,
    DirectMessages,
    SavedMessages,
}

/// Settings tabs.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SettingsTab {
    Account,
    Privacy,
    Appearance,
    Notifications,
    Encryption,
}

/// Toast notification levels.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ToastLevel {
    Info,
    Success,
    Warning,
    Error,
}
```

Import ChannelId, MessageId from domain, ModalType from state::ui_state.
