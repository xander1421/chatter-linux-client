2. Add helper method to get profile key:
```rust
fn get_profile_key(&self) -> Result<[u8; 32], SavedMessageError> {
    let key_b64 = self.settings_repo
        .get_string(keys::PROFILE_KEY)
        .map_err(|_| SavedMessageError::NotLoggedIn)?
        .ok_or(SavedMessageError::NotLoggedIn)?;

    let key_bytes = base64::engine::general_purpose::STANDARD
        .decode(&key_b64)
        .map_err(|_| SavedMessageError::InvalidData("invalid profile key".to_string()))?;

    if key_bytes.len() != 32 {
        return Err(SavedMessageError::InvalidData("profile key wrong length".to_string()));
    }

    let mut key = [0u8; 32];
    key.copy_from_slice(&key_bytes);
    Ok(key)
}
```

3. Add high-level encryption helpers:
```rust
/// Encrypt text content for saving.
pub fn encrypt_text(&self, text: &str) -> Result<SavedMessageContent, SavedMessageError> {
    let profile_key = self.get_profile_key()?;
    SavedMessageCrypto::encrypt_text(text, &profile_key)
        .map_err(SavedMessageError::Encryption)
}

/// Encrypt source metadata for saving.
pub fn encrypt_source(&self, source: &SavedMessageSourcePlain) -> Result<SavedMessageSource, SavedMessageError> {
    let profile_key = self.get_profile_key()?;
    SavedMessageCrypto::encrypt_source(source, &profile_key)
        .map_err(SavedMessageError::Encryption)
}

/// Decrypt message content after fetching.
pub fn decrypt_content(&self, content: &SavedMessageContent) -> Result<String, SavedMessageError> {
    let profile_key = self.get_profile_key()?;
    SavedMessageCrypto::decrypt_content(content, &profile_key)
        .map_err(SavedMessageError::Encryption)
}

/// Decrypt source metadata after fetching.
pub fn decrypt_source(&self, source: &SavedMessageSource) -> Result<SavedMessageSourcePlain, SavedMessageError> {
    let profile_key = self.get_profile_key()?;
    SavedMessageCrypto::decrypt_source(source, &profile_key)
        .map_err(SavedMessageError::Encryption)
}
```
