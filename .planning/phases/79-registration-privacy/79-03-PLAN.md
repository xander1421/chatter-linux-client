  ```rust
  //! Friend request message encryption.
  //!
  //! Encrypts friend request messages using Signal Protocol before sending.
  //! Establishes session with recipient if not already established.

  use std::sync::Arc;
  use crate::domain::ids::UserId;
  use crate::error::ServiceError;
  use crate::services::signal::SignalService;
  use crate::network::grpc::auth_client::AuthClient;

  pub struct FriendRequestCrypto {
      signal_service: Arc<SignalService>,
  }

  impl FriendRequestCrypto {
      pub fn new(signal_service: Arc<SignalService>) -> Self {
          Self { signal_service }
      }

      /// Encrypt friend request message for recipient.
      ///
      /// 1. Fetch recipient's prekey bundle if no session exists
      /// 2. Establish Signal Protocol session
      /// 3. Encrypt message using Double Ratchet
      /// 4. Return encrypted bytes
      pub async fn encrypt_message(
          &self,
          auth_client: &mut AuthClient,
          recipient_id: &UserId,
          message: &str,
      ) -> Result<Vec<u8>, ServiceError> {
          // Check if session exists, if not establish one
          let session_exists = self.signal_service
              .has_session(recipient_id, 1)  // device_id 1 for primary
              .await?;

          if !session_exists {
              // Fetch prekey bundle
              let bundle_response = auth_client
                  .get_pre_key_bundle(&recipient_id.to_string(), "*")
                  .await?;

              // Establish session with prekey bundle
              if let Some(result) = bundle_response.result {
                  for device_bundle in result.device_bundles {
                      self.signal_service
                          .process_prekey_bundle(recipient_id, &device_bundle)
                          .await?;
                  }
              }
          }

          // Encrypt message
          let plaintext = message.as_bytes();
          let encrypted = self.signal_service
              .encrypt_message(recipient_id, 1, plaintext)
              .await?;

          Ok(encrypted)
      }

      /// Decrypt friend request message from sender.
      pub async fn decrypt_message(
          &self,
          sender_id: &UserId,
          device_id: u32,
          ciphertext: &[u8],
      ) -> Result<String, ServiceError> {
          let plaintext = self.signal_service
              .decrypt_message(sender_id, device_id, ciphertext)
              .await?;

          String::from_utf8(plaintext)
              .map_err(|_| ServiceError::Crypto("invalid UTF-8 in friend request".into()))
      }
  }
  ```
