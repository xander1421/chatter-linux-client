Create migration v012 with tables for DM mailbox state:

```sql
-- Active DM mailboxes (our mailboxes for receiving)
CREATE TABLE dm_mailboxes (
    id TEXT PRIMARY KEY,                      -- UUID
    conversation_id TEXT NOT NULL UNIQUE,     -- Peer user ID (DM is 1:1)
    mailbox_public_key BLOB NOT NULL,         -- Ed25519 public key (32 bytes)
    mailbox_private_key BLOB NOT NULL,        -- Ed25519 private key (64 bytes)
    signature BLOB NOT NULL,                  -- Unblinded RSA signature
    randomizer BLOB,                          -- Message randomizer (optional)
    token_id TEXT NOT NULL,                   -- Original token ID (for tracking)
    created_at TEXT NOT NULL,
    expires_at TEXT NOT NULL,                 -- When server will delete
    last_polled_at TEXT,                      -- For polling optimization
    FOREIGN KEY (token_id) REFERENCES mailbox_tokens(id)
);

-- Peer mailbox addresses (their mailboxes for sending)
CREATE TABLE dm_peer_mailboxes (
    id TEXT PRIMARY KEY,
    conversation_id TEXT NOT NULL,            -- Same peer user ID
    peer_mailbox_public_key BLOB NOT NULL,    -- Their mailbox Ed25519 public key
    received_at TEXT NOT NULL,                -- When we learned this address
    expires_at TEXT,                          -- When they said it expires (if known)
    UNIQUE(conversation_id, peer_mailbox_public_key)
);

-- Shared conversation keys (for message encryption)
CREATE TABLE dm_conversation_keys (
    id TEXT PRIMARY KEY,
    conversation_id TEXT NOT NULL,
    our_ephemeral_public BLOB NOT NULL,       -- Our ephemeral Ed25519 key
    our_ephemeral_private BLOB NOT NULL,
    their_ephemeral_public BLOB,              -- Their key (NULL until received)
    shared_secret BLOB,                       -- X25519 shared secret (NULL until derived)
    key_version INTEGER NOT NULL DEFAULT 1,   -- For key rotation
    created_at TEXT NOT NULL,
    established_at TEXT,                      -- When both keys present
    UNIQUE(conversation_id, key_version)
);
```
