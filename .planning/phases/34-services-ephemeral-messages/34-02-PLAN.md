Create EphemeralService for ephemeral message lifecycle management:

```rust
use std::sync::Arc;
use crate::domain::ephemeral::{EphemeralMessage, MessageTTL};
use crate::domain::ids::MessageId;
use crate::storage::repositories::ephemeral_repo::EphemeralRepository;
use crate::storage::repositories::message_repo::MessageRepository;
use crate::storage::repositories::settings_repo::{keys, SettingsRepository};
use crate::error::StorageError;

#[derive(Debug, thiserror::Error)]
pub enum EphemeralError {
    #[error("not logged in")]
    NotLoggedIn,

    #[error("message not found: {0}")]
    MessageNotFound(MessageId),

    #[error("message is not ephemeral: {0}")]
    NotEphemeral(MessageId),

    #[error("storage error: {0}")]
    Storage(#[from] StorageError),
}

pub struct EphemeralService {
    settings_repo: Arc<SettingsRepository>,
    ephemeral_repo: Arc<EphemeralRepository>,
    message_repo: Arc<MessageRepository>,
}

impl EphemeralService {
    pub fn new(
        settings_repo: Arc<SettingsRepository>,
        ephemeral_repo: Arc<EphemeralRepository>,
        message_repo: Arc<MessageRepository>,
    ) -> Self;

    // Check login status (reuse pattern from MessageService)
    fn check_logged_in(&self) -> Result<(), EphemeralError>;

    // Track a new ephemeral message (called after message sent)
    pub fn track(&self, message_id: MessageId, ttl: MessageTTL) -> Result<EphemeralMessage, EphemeralError>;

    // Mark ephemeral message as read (starts countdown)
    pub fn mark_read(&self, message_id: &MessageId) -> Result<EphemeralMessage, EphemeralError>;

    // Get ephemeral state for a message (if it's ephemeral)
    pub fn get_state(&self, message_id: &MessageId) -> Result<Option<EphemeralMessage>, EphemeralError>;

    // Cleanup expired messages - deletes from both ephemeral_messages and messages tables
    // Returns list of deleted message IDs
    pub fn cleanup_expired(&self) -> Result<Vec<MessageId>, EphemeralError>;

    // List all pending (unread) ephemeral messages
    pub fn list_pending(&self) -> Result<Vec<EphemeralMessage>, EphemeralError>;
}
```

Key implementation details:
- `track()`: Creates EphemeralMessage with TTL, stores via repository
- `mark_read()`: Calls repository mark_read, returns updated EphemeralMessage
- `cleanup_expired()`: Gets expired from repo, deletes both ephemeral tracking AND the message itself (soft delete via message_repo)
- All operations require login check

Export from mod.rs.

/// Send an ephemeral (self-destructing) message to a channel.
///
/// The message will be marked for deletion after the TTL expires,
/// starting from when the recipient reads it.
///
/// # Arguments
///
/// * `message_client` - Mutable reference to message client
/// * `channel_id` - Target channel ID
/// * `text` - Message text content
/// * `ttl` - Time-to-live for the message
/// * `attachment_ids` - Optional attachment IDs
/// * `thread_id` - Optional parent message ID for threading
#[allow(clippy::too_many_arguments)]
pub async fn send_ephemeral_message(
    &self,
    message_client: &mut MessageClient,
    channel_id: &ChannelId,
    text: &str,
    ttl: MessageTTL,
    attachment_ids: Vec<AttachmentId>,
    thread_id: Option<&MessageId>,
) -> Result<Message, MessageError>;
```

Implementation:
1. Check logged in (existing pattern)
2. Validate content (must have text or attachments)
3. Build proto content
4. Convert TTL to prost_types::Duration: `Some(prost_types::Duration { seconds: ttl.as_secs() as i64, nanos: 0 })`
5. Call `message_client.send_message()` with `ephemeral: true` and `ephemeral_ttl: Some(duration)`
6. Convert response to domain Message
7. Store locally via message_repo

NOTE: The caller is responsible for also calling `EphemeralService::track()` after this returns successfully to set up local ephemeral tracking. This keeps services loosely coupled.

Also add import for `MessageTTL` at top of file:

    use super::*;

    #[test]
    fn test_ephemeral_error_display() {
        let err = EphemeralError::NotLoggedIn;
        assert!(err.to_string().contains("not logged in"));

        let msg_id = MessageId::new();
        let err = EphemeralError::MessageNotFound(msg_id.clone());
        assert!(err.to_string().contains("message not found"));

        let err = EphemeralError::NotEphemeral(msg_id);
        assert!(err.to_string().contains("not ephemeral"));
    }

    #[test]
    fn test_message_ttl_to_prost_duration() {
        let ttl = MessageTTL::Minutes5;
        let duration = prost_types::Duration {
            seconds: ttl.as_secs() as i64,
            nanos: 0,
        };
        assert_eq!(duration.seconds, 300);
        assert_eq!(duration.nanos, 0);
    }
}
```

These tests verify:
