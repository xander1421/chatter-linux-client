  <name>Task 1: Extend DomainEvent for reactions and receipts</name>
  <files>src/inbox/handlers/mod.rs</files>
  <action>
Add DomainEvent variants for reactions and receipts:

1. **DomainEvent::ReactionAdded**:
   - `message_id: MessageId` - target message
   - `author_id: UserId` - message author (for lookup)
   - `reactor_id: UserId` - who reacted
   - `emoji: String` - the emoji
   - `timestamp: u64` - when reaction was sent

2. **DomainEvent::ReactionRemoved**:
   - Same fields as ReactionAdded
   - Indicates removal rather than addition

3. **DomainEvent::DeliveryReceipt**:
   - `message_timestamps: Vec<u64>` - messages confirmed delivered
   - `from_user_id: UserId` - recipient who confirmed delivery
   - `receipt_timestamp: u64` - when receipt was sent

4. **DomainEvent::ReadReceipt**:
   - `message_timestamps: Vec<u64>` - messages marked as read
   - `from_user_id: UserId` - user who read
   - `receipt_timestamp: u64` - when read occurred

5. **DomainEvent::ViewedReceipt** (for view-once messages):
   - `message_timestamp: u64` - the viewed message
   - `from_user_id: UserId` - user who viewed
   - `receipt_timestamp: u64` - when viewed

Note: Receipts use timestamps as message identifiers (Signal protocol pattern).

3. **handle() implementation**:
   ```rust
   async fn handle(&self, message: &ProcessedMessage, ctx: &HandlerContext) -> HandlerResult {
       let content: Content = prost::Message::decode(message.plaintext.as_slice())
           .map_err(|e| HandlerError::ParseFailed { error: e.to_string() })?;

       let reaction = match content.content {
           Some(content::Content::Reaction(r)) => r,
           _ => return Ok(HandlerOutcome::Ignored),
       };

       // Validate required fields
       if reaction.emoji.is_empty() {
           return Err(HandlerError::ValidationFailed {
               reason: "reaction emoji is empty".to_string(),
           });
       }

       if reaction.target_author_aci.is_empty() {
           return Err(HandlerError::ValidationFailed {
               reason: "reaction target_author_aci is empty".to_string(),
           });
       }

       // Get reactor from message sender
       let reactor_id = message.sender_address.as_ref()
           .map(|addr| UserId::from_string(addr.name().to_string()))
           .ok_or_else(|| HandlerError::MissingContext {
               field: "sender_address".to_string(),
           })?;

       let author_id = UserId::from_string(reaction.target_author_aci.clone());

       // Emit appropriate event
       let event = if reaction.remove {
           DomainEvent::ReactionRemoved {
               message_id: MessageId::from_timestamp(reaction.target_sent_timestamp),
               author_id,
               reactor_id,
               emoji: reaction.emoji.clone(),
               timestamp: reaction.timestamp,
           }
       } else {
           DomainEvent::ReactionAdded {
               message_id: MessageId::from_timestamp(reaction.target_sent_timestamp),
               author_id,
               reactor_id,
               emoji: reaction.emoji.clone(),
               timestamp: reaction.timestamp,
           }
       };

       let _ = ctx.event_tx.send(event);
       Ok(HandlerOutcome::Processed)
   }
   ```

  <action>
Implement ReceiptHandler for ReceiptMessage:

1. **ReceiptHandler** struct (stateless)

2. **InboxHandler impl**:
   - `content_types()` returns ContentType for receipts (tag 5 in Content oneof)

3. **handle() implementation**:
   ```rust
   async fn handle(&self, message: &ProcessedMessage, ctx: &HandlerContext) -> HandlerResult {
       let content: Content = prost::Message::decode(message.plaintext.as_slice())
           .map_err(|e| HandlerError::ParseFailed { error: e.to_string() })?;

       let receipt = match content.content {
           Some(content::Content::Receipt(r)) => r,
           _ => return Ok(HandlerOutcome::Ignored),
       };

       // Get sender (who sent the receipt)
       let from_user_id = message.sender_address.as_ref()
           .map(|addr| UserId::from_string(addr.name().to_string()))
           .ok_or_else(|| HandlerError::MissingContext {
               field: "sender_address".to_string(),
           })?;

       // Empty timestamps list is technically valid (no-op)
       if receipt.timestamps.is_empty() {
           return Ok(HandlerOutcome::Processed);
       }

       // Dispatch based on receipt type
       let receipt_type = receipt_message::Type::try_from(receipt.r#type)
           .unwrap_or(receipt_message::Type::Unspecified);

       let event = match receipt_type {
           receipt_message::Type::Delivery => DomainEvent::DeliveryReceipt {
               message_timestamps: receipt.timestamps.clone(),
               from_user_id,
               receipt_timestamp: receipt.timestamp,
           },
           receipt_message::Type::Read => DomainEvent::ReadReceipt {
               message_timestamps: receipt.timestamps.clone(),
               from_user_id,
               receipt_timestamp: receipt.timestamp,
           },
           receipt_message::Type::Viewed => {
               // Viewed receipts are typically for single messages
               for ts in &receipt.timestamps {
                   let _ = ctx.event_tx.send(DomainEvent::ViewedReceipt {
                       message_timestamp: *ts,
                       from_user_id: from_user_id.clone(),
                       receipt_timestamp: receipt.timestamp,
                   });
               }
               return Ok(HandlerOutcome::Processed);
           }
           receipt_message::Type::Unspecified => {
               tracing::warn!("Received receipt with unspecified type, ignoring");
               return Ok(HandlerOutcome::Ignored);
           }
       };

       let _ = ctx.event_tx.send(event);
       Ok(HandlerOutcome::Processed)
   }
   ```
