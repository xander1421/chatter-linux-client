   - `content_types()` returns `[ContentType::TypingMessage]`

5. **handle() implementation**:
   ```rust
   async fn handle(&self, message: &ProcessedMessage, ctx: &HandlerContext) -> HandlerResult {
       let content: Content = prost::Message::decode(message.plaintext.as_slice())
           .map_err(|e| HandlerError::ParseFailed { error: e.to_string() })?;

       let typing = match content.content {
           Some(content::Content::Typing(t)) => t,
           _ => return Ok(HandlerOutcome::Ignored),
       };

       // Get user who is typing
       let user_id = message.sender_address.as_ref()
           .map(|addr| UserId::from_string(addr.name().to_string()))
           .ok_or_else(|| HandlerError::MissingContext {
               field: "sender_address".to_string(),
           })?;

       // Parse channel_id (may be empty for DMs)
       let channel_id = if typing.channel_id.is_empty() {
           // For DMs, derive channel from sender
           ChannelId::from_dm_participants(&user_id, &ctx.local_user_id)
       } else {
           ChannelId::from_string(typing.channel_id.clone())
       };

       // Dispatch based on action
       let action = typing_message::Action::try_from(typing.action)
           .unwrap_or(typing_message::Action::Unspecified);

       let event = match action {
           typing_message::Action::Started => DomainEvent::TypingStarted {
               channel_id,
               user_id,
               timestamp: typing.timestamp,
           },
           typing_message::Action::Stopped => DomainEvent::TypingStopped {
               channel_id,
               user_id,
               timestamp: typing.timestamp,
           },
           typing_message::Action::Unspecified => {
               tracing::debug!("Received typing indicator with unspecified action");
               return Ok(HandlerOutcome::Ignored);
           }
       };

       let _ = ctx.event_tx.send(event);
       Ok(HandlerOutcome::Processed)
   }
   ```

   - Sort user IDs and concatenate/hash for consistency
   - Add to domain/ids.rs if needed

   ```rust
           let context = HandlerContext {
               local_user_id,
               event_tx,
           };

           let handlers: Vec<Arc<dyn InboxHandler>> = vec![

               Arc::new(AttachmentHandler),
               Arc::new(ReactionHandler),

               Arc::new(TypingHandler),
           ];

           Self { handlers, context }
       }
   }
   ```

3. **route() method**:
   ```rust
   /// Route a processed message to the appropriate handler.
   pub async fn route(&self, message: &ProcessedMessage) -> HandlerResult {
       // Find handler that supports this content type
       for handler in &self.handlers {
           if handler.content_types().contains(&message.content_type) {
               return handler.handle(message, &self.context).await;
           }
       }

       // No handler found for content type
       tracing::debug!(
           content_type = %message.content_type,
           "No handler registered for content type"
       );
       Ok(HandlerOutcome::Ignored)
   }
   ```

4. **register() method** for adding custom handlers:
   ```rust
   /// Register an additional handler.
   pub fn register(&mut self, handler: Arc<dyn InboxHandler>) {
       self.handlers.push(handler);
   }
   ```

5. **subscribe() method** for getting event receiver:
   ```rust
   /// Subscribe to domain events.
   pub fn subscribe(&self) -> broadcast::Receiver<DomainEvent> {
       self.context.event_tx.subscribe()
   }
   ```
