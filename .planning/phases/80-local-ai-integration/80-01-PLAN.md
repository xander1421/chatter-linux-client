impl LocalAIActorHandle {
    pub async fn load_model(&self, model_id: String, quantized_file: Option<String>)
        -> Result<ModelInfo, LocalAIError>
    {
        let (tx, rx) = oneshot::channel();
        self.sender.send(LocalAICommand::LoadModel {
            model_id, quantized_file, response: tx
        }).await.map_err(|_| LocalAIError::ActorDead)?;
        rx.await.map_err(|_| LocalAIError::ActorDead)?
    }

    pub async fn complete(&self, prompt: String, params: SampleParams)
        -> Result<String, LocalAIError> { ... }
