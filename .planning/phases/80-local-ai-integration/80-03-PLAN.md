Add convenience methods:

```rust
impl SettingsRepository {
    pub fn get_local_ai_enabled(&self) -> Result<bool, StorageError> {
        Ok(self.get_string(keys::LOCAL_AI_ENABLED)?
            .map(|s| s == "true")
            .unwrap_or(false))
    }

    pub fn set_local_ai_enabled(&self, enabled: bool) -> Result<(), StorageError> {
        self.set_string(keys::LOCAL_AI_ENABLED, &enabled.to_string())
    }

    pub fn get_local_ai_model(&self) -> Result<Option<(String, Option<String>)>, StorageError> {
        let model_id = self.get_string(keys::LOCAL_AI_MODEL_ID)?;
        let quantized = self.get_string(keys::LOCAL_AI_QUANTIZED_FILE)?;
        Ok(model_id.map(|id| (id, quantized)))
    }

    pub fn set_local_ai_model(&self, model_id: &str, quantized_file: Option<&str>) -> Result<(), StorageError> {
        self.set_string(keys::LOCAL_AI_MODEL_ID, model_id)?;
        if let Some(qf) = quantized_file {
            self.set_string(keys::LOCAL_AI_QUANTIZED_FILE, qf)?;
        } else {
            self.delete(keys::LOCAL_AI_QUANTIZED_FILE)?;
        }
        Ok(())
    }
}
```

Create LocalAIService that wraps the actor and provides high-level operations:

```rust
use std::sync::Arc;
use crate::storage::repositories::settings_repo::SettingsRepository;
use super::{LocalAIActorHandle, LocalAIError, SampleParams, ModelInfo};

pub struct LocalAIService {
    #[cfg(feature = "local-ai")]
    actor: LocalAIActorHandle,
    settings_repo: Arc<SettingsRepository>,
}

impl LocalAIService {
    #[cfg(feature = "local-ai")]
    pub fn new(actor: LocalAIActorHandle, settings_repo: Arc<SettingsRepository>) -> Self {
        Self { actor, settings_repo }
    }

    #[cfg(not(feature = "local-ai"))]
    pub fn new(settings_repo: Arc<SettingsRepository>) -> Self {
        Self { settings_repo }
    }

    /// Load the user's preferred model from settings.
    #[cfg(feature = "local-ai")]
    pub async fn load_preferred_model(&self) -> Result<Option<ModelInfo>, LocalAIError> {
        if !self.settings_repo.get_local_ai_enabled().unwrap_or(false) {
            return Ok(None);
        }
        if let Ok(Some((model_id, quantized))) = self.settings_repo.get_local_ai_model() {
            let info = self.actor.load_model(model_id, quantized).await?;
            Ok(Some(info))
        } else {
            Ok(None)
        }
    }

    /// Summarize a conversation.
    #[cfg(feature = "local-ai")]
    pub async fn summarize(&self, messages: &[String]) -> Result<String, LocalAIError> {
        let prompt = format!(
            "Summarize this conversation in 2-3 sentences:\n\n{}",
            messages.join("\n")
        );
        self.actor.complete(prompt, SampleParams::default()).await
    }

    /// Generate a smart reply suggestion.
    #[cfg(feature = "local-ai")]
    pub async fn smart_reply(&self, context: &str, last_message: &str) -> Result<String, LocalAIError> {
        let prompt = format!(
            "Given this conversation context:\n{}\n\nSuggest a brief, natural reply to: \"{}\"\n\nReply:",
            context, last_message
        );
        let mut params = SampleParams::default();
        params.max_tokens = 100;
        self.actor.complete(prompt, params).await
    }

    /// Check if local AI is available and enabled.
    pub fn is_available(&self) -> bool {
        #[cfg(feature = "local-ai")]
        { self.settings_repo.get_local_ai_enabled().unwrap_or(false) }
        #[cfg(not(feature = "local-ai"))]
        { false }
    }
}
```
