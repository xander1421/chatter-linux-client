```rust
#[cfg(feature = "local-ai")]
use mistralrs::{
    TextModelBuilder, GgufModelBuilder, Device, MistralRs,
    Request, RequestMessage, Response, SamplingParams,
};

impl LocalAIActorHandle {
    pub fn spawn() -> Self {
        let (sender, mut receiver) = mpsc::channel(32);

        std::thread::spawn(move || {
            let runtime = tokio::runtime::Builder::new_current_thread()
                .enable_all()
                .build()
                .expect("Local AI runtime");

            runtime.block_on(async move {
                #[cfg(feature = "local-ai")]
                let mut pipeline: Option<Arc<MistralRs>> = None;

                #[cfg(not(feature = "local-ai"))]
                let mut _model_info: Option<ModelInfo> = None;

                while let Some(cmd) = receiver.recv().await {
                    match cmd {
                        LocalAICommand::LoadModel { model_id, quantized_file, response } => {
                            #[cfg(feature = "local-ai")]
                            {
                                let result = load_model_impl(&model_id, quantized_file.as_deref()).await;
                                match result {
                                    Ok(p) => {
                                        let info = ModelInfo {
                                            model_id: model_id.clone(),
                                            quantization: quantized_file,
                                            context_length: 8192,
                                            loaded_at: std::time::Instant::now(),
                                        };
                                        pipeline = Some(p);
                                        let _ = response.send(Ok(info));
                                    }
                                    Err(e) => {
                                        let _ = response.send(Err(LocalAIError::LoadFailed(e.to_string())));
                                    }
                                }
                            }
                            #[cfg(not(feature = "local-ai"))]
                            {
                                let _ = response.send(Err(LocalAIError::LoadFailed(
                                    "local-ai feature not enabled".to_string()
                                )));
                            }
                        }
                        // ... other commands
                    }
                }
            });
        });

        Self { sender }
    }
}

#[cfg(feature = "local-ai")]
async fn load_model_impl(model_id: &str, quantized_file: Option<&str>) -> anyhow::Result<Arc<MistralRs>> {
    // Use GgufModelBuilder for GGUF models, TextModelBuilder for safetensors
    if let Some(gguf) = quantized_file {
        GgufModelBuilder::new(model_id, vec![gguf.to_string()])
            .with_device(Device::Cuda(0))
            .build()
            .await
    } else {
        TextModelBuilder::new(model_id)
            .with_device(Device::Cuda(0))
            .build()
            .await
    }
}
```
